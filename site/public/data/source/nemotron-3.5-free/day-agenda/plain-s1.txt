export type TagColor = 'blue' | 'green' | 'amber' | 'red' | 'purple' | 'gray';

export interface EventTag {
  label: string;
  color: TagColor;
}

export interface AgendaEvent {
  id: string;
  start: string;
  end: string;
  title: string;
  location: string;
  attendees: string[];
  tag: EventTag;
}

export interface Props {
  dateLabel: string;
  events: AgendaEvent[];
  addEventLabel: string;
}

function tagBadgeClasses(color: TagColor): string {
  switch (color) {
    case 'blue':
      return 'bg-blue-100 text-blue-700 border-blue-200';
    case 'green':
      return 'bg-green-100 text-green-700 border-green-200';
    case 'amber':
      return 'bg-amber-100 text-amber-700 border-amber-200';
    case 'red':
      return 'bg-red-100 text-red-700 border-red-200';
    case 'purple':
      return 'bg-purple-100 text-purple-700 border-purple-200';
    case 'gray':
      return 'bg-gray-100 text-gray-700 border-gray-200';
  }
}

export default function DayAgenda({ dateLabel, events, addEventLabel }: Props) {
  return (
    <div className="w-full max-w-md border border-gray-200 rounded-lg shadow-sm bg-white">
      <div className="flex items-center justify-between px-4 py-3 border-b border-gray-200">
        <span className="text-base font-semibold text-gray-900">{dateLabel}</span>
        <button
          type="button"
          className="inline-flex items-center gap-1.5 rounded-md border border-transparent bg-blue-50 px-3 py-1.5 text-sm font-medium text-blue-700 shadow-sm hover:bg-blue-100 focus:outline-none focus:ring-2 focus:ring-blue-500 focus:ring-offset-2"
        >
          {addEventLabel}
        </button>
      </div>
      <div className="divide-y divide-gray-100">
        {events.map((event) => (
          <div key={event.id} className="px-4 py-3">
            <div className="flex items-start gap-3">
              <div className="flex flex-col items-center min-w-[48px] text-sm text-gray-500">
                <span className="font-medium">{event.start}</span>
                <span className="mt-0.5 text-xs">{event.end}</span>
              </div>
              <div className="flex-1">
                <div className="flex items-center gap-2 mb-1">
                  <span
                    className={`inline-flex items-center rounded-md border px-2 py-0.5 text-xs font-medium border-transparent ${tagBadgeClasses(event.tag.color)}`}
                  >
                    {event.tag.label}
                  </span>
                </div>
                <h3 className="text-sm font-semibold text-gray-900">{event.title}</h3>
                <p className="text-xs text-gray-500">{event.location}</p>
                <p className="mt-1 text-xs text-gray-500">{event.attendees.join(', ')}</p>
              </div>
            </div>
          </div>
        ))}
      </div>
    </div>
  );
}
