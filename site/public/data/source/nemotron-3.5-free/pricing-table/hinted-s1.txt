export interface Plan {
  name: string;
  monthlyPrice: number;
  description: string;
  features: string[];
  ctaLabel: string;
  badge?: string;
  highlighted: boolean;
}

export interface Props {
  heading: string;
  subheading: string;
  currency: string;
  plans: Plan[];
}

export default function PricingTable({ heading, subheading, currency, plans }: Props) {
  return (
    <section className="bg-gray-50">
      <div className="mx-auto max-w-7xl px-4 sm:px-6 lg:px-8">
        <div className="mx-auto max-w-2xl text-center">
          <h2 className="text-3xl font-bold tracking-tight text-gray-900 sm:text-4xl">{heading}</h2>
          <p className="mt-4 text-lg text-gray-600">{subheading}</p>
        </div>
        <div className="mt-10 sm:mt-12 grid grid-cols-1 gap-y-10 gap-x-6 sm:grid-cols-3 sm:gap-y-0 sm:gap-x-8">
          {plans.map((plan, index) => (
            <div
              key={index}
              className={`flex flex-col rounded-xl p-6 sm:p-8 min-h-full ${
                plan.highlighted
                  ? "ring-2 ring-indigo-600 bg-indigo-50 border border-indigo-200 shadow-xl"
                  : "border border-gray-200 bg-white shadow-sm"
              }`}
            >
              <div className="flex items-center justify-between gap-2 mb-2 sm:mb-3">
                <div className="min-w-0 flex flex-col">
                  {plan.highlighted && (
                    <span className="inline-flex items-center rounded-md bg-indigo-100 px-2.5 py-0.5 text-xs font-semibold text-indigo-800">
                      Recommended
                    </span>
                  )}
                  <h3 className="text-lg sm:text-xl font-semibold text-gray-900">{plan.name}</h3>
                </div>
                {plan.badge && (
                  <span className="shrink-0 inline-flex items-center rounded-full border border-indigo-200 bg-white px-2.5 py-0.5 text-xs font-semibold text-indigo-700 shadow-sm ring-1 ring-transparent">
                    {plan.badge}
                  </span>
                )}
              </div>
              <div className="mt-2 sm:mt-3 flex items-baseline">
                <span className="text-3xl sm:text-4xl font-bold text-gray-900">
                  {currency} {plan.monthlyPrice}
                </span>
                <span className="ml-1 text-sm text-gray-500">/mo</span>
              </div>
              <p className="mt-3 sm:mt-4 text-sm text-gray-600">{plan.description}</p>
              <ul className="mt-6 sm:mt-8 space-y-3 sm:space-y-4 list-none">
                {plan.features.map((feature, featureIndex) => (
                  <li key={featureIndex} className="flex items-start gap-2 text-sm text-gray-700">
                    <svg className="w-5 h-5 shrink-0 text-indigo-600" viewBox="0 0 20 20" fill="currentColor" aria-hidden="true">
                      <path fillRule="evenodd" d="M16.707 5.293a1 1 0 010 1.414l-8 8a1 1 0 01-1.414 0l-4-4a1 1 0 011.414-1.414L8 12.586l7.293-7.293a1 1 0 011.414 0z" clipRule="evenodd" />
                    </svg>
                    {feature}
                  </li>
                ))}
              </ul>
              <button
                type="button"
                className={`mt-auto sm:mt-10 flex w-full justify-center rounded-md px-4 py-2 text-sm font-semibold shadow-sm ring-1 ring-inset whitespace-normal break-words ${
                  plan.highlighted
                    ? "bg-indigo-600 text-white hover:bg-indigo-700"
                    : "bg-transparent text-gray-900 ring-gray-300 hover:bg-gray-50"
                }`}
              >
                {plan.ctaLabel}
              </button>
            </div>
          ))}
        </div>
      </div>
    </section>
  );
}
