export interface Props {
  heading: string;
  subheading: string;
  currency: string;
  plans: Plan[];
}

export interface Plan {
  name: string;
  monthlyPrice: number;
  description: string;
  features: string[];
  ctaLabel: string;
  badge?: string;
  highlighted: boolean;
}

export default function PricingSection({ heading, subheading, currency, plans }: Props) {
  return (
    <section className="py-20 px-4 sm:px-6 lg:px-8">
      <div className="max-w-7xl mx-auto">
        <div className="text-center max-w-3xl mx-auto mb-16">
          <h2 className="text-3xl font-bold tracking-tight text-gray-900 sm:text-4xl">{heading}</h2>
          <p className="mt-4 text-lg text-gray-600">{subheading}</p>
        </div>

        <div className="grid gap-y-12 gap-x-8 sm:grid-cols-2 lg:grid-cols-3">
          {plans.map((plan) => (
            <article
              key={plan.name}
              className={`
                relative rounded-2xl border p-8 flex flex-col
                ${plan.highlighted
                  ? "border-indigo-600 bg-indigo-50 shadow-xl ring-1 ring-indigo-600"
                  : "border-gray-200 bg-white shadow-sm"
              }
              `}
            >
              {plan.badge && (
                <span className={`absolute top-4 left-1/2 -translate-x-1/2 z-10 rounded-full border px-3 py-0.5 text-xs font-semibold uppercase tracking-tighter ${
                  plan.highlighted
                    ? "border-indigo-600 bg-indigo-100 text-indigo-700"
                    : "border-gray-200 bg-gray-100 text-gray-700"
                }`}>
                  {plan.badge}
                </span>
              )}

              <div className="relative z-10 mt-4">
                <h3 className="text-xl font-bold text-gray-900">{plan.name}</h3>
                <div className="mt-1 flex items-baseline gap-1">
                  <span className="text-4xl font-extrabold text-gray-900">{plan.monthlyPrice}</span>
                  <span className="ml-1 text-sm font-semibold text-gray-500">/mo</span>
                  <span className="ml-1 text-xs font-semibold text-gray-400 tracking-wider">{currency}</span>
                </div>
                <p className="mt-4 text-sm text-gray-600 leading-relaxed">{plan.description}</p>
              </div>

              <div className="mt-6 flex-1 space-y-3">
                {plan.features.map((feature, index) => (
                  <div key={index} className="flex items-start">
                    <svg
                      className="mt-0.5 h-5 w-5 shrink-0"
                      fill="none"
                      viewBox="0 0 24 24"
                      stroke="currentColor"
                      aria-hidden="true"
                    >
                      <path
                        strokeLinecap="round"
                        strokeLinejoin="round"
                        strokeWidth="2.5"
                        d="M5 13l4 4L19 7"
                      />
                    </svg>
                    <span className={`ml-3 text-sm leading-6 ${plan.highlighted ? "text-gray-900" : "text-gray-700"}`}>
                      {feature}
                    </span>
                  </div>
                ))}
              </div>

              <div className="mt-auto pt-4">
                <button
                  type="button"
                  className={`w-full rounded-xl py-3 font-semibold text-center shadow-sm ${
                    plan.highlighted
                      ? "bg-indigo-600 text-white hover:bg-indigo-700 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-indigo-600 focus-visible:ring-offset-2"
                      : "bg-white text-gray-900 border border-gray-300 hover:bg-gray-50 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-gray-400 focus-visible:ring-offset-2"
                  }`}
                >
                  {plan.ctaLabel}
                </button>
              </div>
            </article>
          ))}
        </div>
      </div>
    </section>
  );
}
