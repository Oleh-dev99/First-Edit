import { type FC } from "react";

export interface Props {
  title: string;
  currency: string;
  items: OrderItem[];
  promo: PromoField;
  subtotal: number;
  shipping: number;
  tax: number;
  total: number;
  note: string;
  payLabel: string;
}

export interface OrderItem {
  id: string;
  image: ItemImage;
  name: string;
  variant: string;
  quantity: number;
  price: number;
}

export interface ItemImage {
  src: string;
  alt: string;
}

export interface PromoField {
  label: string;
  placeholder: string;
  applyLabel: string;
}

const OrderSummary: FC<Props> = ({
  title,
  currency,
  items,
  promo,
  subtotal,
  shipping,
  tax,
  total,
  note,
  payLabel,
}) => {
  const formatAmount = (amount: number): string => {
    return `${currency} ${amount.toFixed(2)}`;
  };

  return (
    <section className="w-full max-w-md rounded-xl border border-gray-200 bg-white p-6">
      <h2 className="text-base font-semibold text-gray-900">{title}</h2>

      <div className="mt-5 space-y-4">
        {items.length === 0 ? (
          <p className="text-sm text-gray-500">No items in this order.</p>
        ) : (
          items.map((item) => (
            <div key={item.id} className="flex items-start gap-3">
              <img
                src={item.image.src}
                alt={item.image.alt}
                className="w-12 h-12 flex-shrink-0 rounded-md object-cover"
              />
              <div className="flex-1 min-w-0">
                <h3 className="text-sm font-medium text-gray-900 break-words">
                  {item.name}
                </h3>
                <p className="text-xs text-gray-500 break-words">
                  {item.variant}
                </p>
              </div>
              <div className="min-w-[60px]">
                <p className="text-xs text-gray-500">{`${item.quantity} ×`}</p>
                <p className="text-sm font-medium text-gray-900 break-words text-right">
                  {formatAmount(item.price)}
                </p>
              </div>
            </div>
          ))
        )}
      </div>

      <div className="mt-6">
        <label className="block text-sm font-medium text-gray-700">
          {promo.label}
        </label>
        <div className="mt-2 flex flex-col sm:flex-row gap-2">
          <input
            type="text"
            placeholder={promo.placeholder}
            className="flex-1 min-w-0 w-full px-3 py-2 text-sm border border-gray-300 rounded-md outline-none focus-visible:ring-1 focus-visible:ring-blue-500 focus-visible:border-blue-500"
          />
          <button
            type="button"
            className="w-full sm:w-auto px-4 py-2 text-sm font-medium text-white bg-gray-900 rounded-md hover:bg-gray-800"
          >
            {promo.applyLabel}
          </button>
        </div>
      </div>

      <div className="mt-6 space-y-2">
        <div className="flex flex-wrap justify-between gap-2 text-sm">
          <span className="text-gray-600">Subtotal</span>
          <span className="text-gray-900">{formatAmount(subtotal)}</span>
        </div>
        <div className="flex flex-wrap justify-between gap-2 text-sm">
          <span className="text-gray-600">Shipping</span>
          <span className="text-gray-900">{formatAmount(shipping)}</span>
        </div>
        <div className="flex flex-wrap justify-between gap-2 text-sm">
          <span className="text-gray-600">Tax</span>
          <span className="text-gray-900">{formatAmount(tax)}</span>
        </div>
        <div className="mt-3 pt-3 border-t border-gray-100 flex flex-wrap justify-between gap-2 text-base font-bold text-gray-900">
          <span>Total</span>
          <span className="text-gray-900">{formatAmount(total)}</span>
        </div>
      </div>

      {note && (
        <p className="mt-6 text-xs text-gray-500 leading-relaxed break-words">
          {note}
        </p>
      )}

      <button
        type="button"
        className="mt-6 w-full flex items-center justify-center gap-2 px-4 py-3 text-sm font-semibold text-white bg-green-700 rounded-md break-words hover:bg-green-800"
      >
        {payLabel}
      </button>
    </section>
  );
};

export default OrderSummary;
