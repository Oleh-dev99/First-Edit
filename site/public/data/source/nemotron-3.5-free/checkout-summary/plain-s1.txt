import { type ReactElement } from "react";

export interface Props {
  title: string;
  currency: string;
  items: OrderItem[];
  promo: PromoField;
  subtotal: number;
  shipping: number;
  tax: number;
  total: number;
  note: string;
  payLabel: string;
}

export interface OrderItem {
  id: string;
  image: ItemImage;
  name: string;
  variant: string;
  quantity: number;
  price: number;
}

export interface ItemImage {
  src: string;
  alt: string;
}

export interface PromoField {
  label: string;
  placeholder: string;
  applyLabel: string;
}

export default function OrderSummary({
  title,
  currency,
  items,
  promo,
  subtotal,
  shipping,
  tax,
  total,
  note,
  payLabel,
}: Props): ReactElement {
  const format = (amount: number): string => `${currency} ${amount.toFixed(2)}`;

  return (
    <div className="max-w-sm flex-shrink-0 border border-slate-200 bg-white rounded-xl p-6 shadow-sm">
      <h2 className="text-base font-semibold text-slate-900">{title}</h2>

      <div className="mt-4 space-y-3">
        {items.map((item) => (
          <div key={item.id} className="flex items-center gap-3">
            <img
              src={item.image.src}
              alt={item.image.alt}
              className="w-12 h-12 rounded-md bg-slate-100 object-cover flex-shrink-0"
            />
            <div className="flex-1 min-w-0">
              <div className="text-sm font-medium text-slate-900 truncate">
                {item.name}
              </div>
              <div className="text-xs text-slate-500 truncate">
                {item.variant}
              </div>
            </div>
            <div className="text-right hidden sm:block">
              <div className="text-xs text-slate-500">Qty {item.quantity}</div>
              <div className="text-sm font-medium text-slate-900">
                {format(item.price)}
              </div>
            </div>
            <div className="text-right text-sm font-medium text-slate-900 sm:hidden">
              {format(item.price)}
            </div>
          </div>
        ))}
      </div>

      <div className="mt-5">
        <div className="flex items-center justify-between mb-2">
          <label className="text-xs font-semibold text-slate-500 uppercase tracking-wider">
            {promo.label}
          </label>
        </div>
        <div className="flex gap-2">
          <input
            type="text"
            placeholder={promo.placeholder}
            className="flex-1 border border-slate-200 rounded-md px-2 py-2 text-sm text-slate-900 placeholder-slate-400 focus:outline-none focus:ring-1 focus:ring-slate-900"
          />
          <button type="button" className="bg-slate-900 text-white text-sm px-3 py-2 rounded-md">
            {promo.applyLabel}
          </button>
        </div>
      </div>

      <div className="mt-5 space-y-2">
        <div className="flex justify-between text-sm">
          <span className="text-slate-500">Subtotal</span>
          <span className="text-slate-900">{format(subtotal)}</span>
        </div>
        <div className="flex justify-between text-sm">
          <span className="text-slate-500">Shipping</span>
          <span className="text-slate-900">{format(shipping)}</span>
        </div>
        <div className="flex justify-between text-sm">
          <span className="text-slate-500">Tax</span>
          <span className="text-slate-900">{format(tax)}</span>
        </div>
        <div className="mt-2 pt-2 border-t border-slate-200 flex justify-between text-base font-bold text-slate-900">
          <span>Total</span>
          <span>{format(total)}</span>
        </div>
      </div>

      <p className="mt-4 text-xs text-slate-500 leading-relaxed">{note}</p>

      <button
        type="button"
        className="mt-5 w-full bg-slate-900 text-white text-sm font-medium py-3 rounded-md"
      >
        {payLabel}
      </button>
    </div>
  );
}
