interface Props {
  invoiceNumber: string;
  issueDate: string;
  dueDate: string;
  currency: string;
  seller: Party;
  buyer: Party;
  lineItems: LineItem[];
  subtotal: number;
  taxLabel: string;
  tax: number;
  total: number;
  note: string;
  payLabel: string;
}

interface Party {
  name: string;
  address: Address;
}

interface Address {
  street: string;
  city: string;
  postalCode: string;
  country: string;
}

interface LineItem {
  id: string;
  description: string;
  quantity: number;
  unitPrice: number;
  amount: number;
}

function formatAmount(currency: string, value: number): string {
  try {
    return new Intl.NumberFormat(undefined, {
      style: "currency",
      currency,
    }).format(value);
  } catch {
    return `${currency} ${value.toFixed(2)}`;
  }
}

export default function Invoice(props: Props) {
  const { currency } = props;

  return (
    <div className="min-h-screen flex items-center justify-center bg-gray-50 p-4">
      <div className="max-w-[700px] w-full max-h-[90vh] overflow-y-auto bg-white rounded-2xl border border-gray-200 shadow-xl p-6 md:p-8">
        <div className="mb-6">
          <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-2">
            <div>
              <h2 className="text-lg font-bold text-gray-900 break-words">
                Invoice {props.invoiceNumber}
              </h2>
              <p className="text-sm text-gray-600 break-words">
                Issue date: {props.issueDate}
              </p>
              <p className="text-sm text-gray-600 break-words">
                Due date: {props.dueDate}
              </p>
            </div>
          </div>
        </div>

        <div className="grid grid-cols-1 md:grid-cols-2 gap-6 mb-6">
          <div className="space-y-1">
            <h3 className="text-xs font-semibold text-gray-500 uppercase tracking-wider">
              From
            </h3>
            <p className="text-base font-medium text-gray-900 break-words">
              {props.seller.name}
            </p>
            <p className="text-sm text-gray-600 break-words">
              {props.seller.address.street}
            </p>
            <p className="text-sm text-gray-600 break-words">
              {props.seller.address.city}, {props.seller.address.postalCode}
            </p>
            <p className="text-sm text-gray-600 break-words">
              {props.seller.address.country}
            </p>
          </div>
          <div className="space-y-1">
            <h3 className="text-xs font-semibold text-gray-500 uppercase tracking-wider">
              Bill to
            </h3>
            <p className="text-base font-medium text-gray-900 break-words">
              {props.buyer.name}
            </p>
            <p className="text-sm text-gray-600 break-words">
              {props.buyer.address.street}
            </p>
            <p className="text-sm text-gray-600 break-words">
              {props.buyer.address.city}, {props.buyer.address.postalCode}
            </p>
            <p className="text-sm text-gray-600 break-words">
              {props.buyer.address.country}
            </p>
          </div>
        </div>

        <div className="overflow-hidden border border-gray-200 rounded-lg mb-6">
          <table className="w-full text-sm text-left">
            <thead className="bg-gray-50 text-gray-600">
              <tr>
                <th className="px-4 py-3 font-medium">Description</th>
                <th className="px-4 py-3 font-medium text-right">Qty</th>
                <th className="px-4 py-3 font-medium text-right">
                  Unit price
                </th>
                <th className="px-4 py-3 font-medium text-right">Amount</th>
              </tr>
            </thead>
            <tbody className="divide-y divide-gray-100">
              {props.lineItems.map((item) => (
                <tr key={item.id}>
                  <td className="px-4 py-3 break-words">
                    {item.description}
                  </td>
                  <td className="px-4 py-3 text-right break-words">
                    {item.quantity}
                  </td>
                  <td className="px-4 py-3 text-right break-words">
                    {formatAmount(currency, item.unitPrice)}
                  </td>
                  <td className="px-4 py-3 text-right font-medium break-words">
                    {formatAmount(currency, item.amount)}
                  </td>
                </tr>
              ))}
              {props.lineItems.length === 0 && (
                <tr>
                  <td colSpan={4} className="px-4 py-4 text-center text-gray-500">
                    No line items
                  </td>
                </tr>
              )}
            </tbody>
          </table>
        </div>

        <div className="space-y-2 mb-6">
          <div className="flex justify-between text-sm">
            <span className="text-gray-600">Subtotal</span>
            <span className="text-gray-900">
              {formatAmount(currency, props.subtotal)}
            </span>
          </div>
          <div className="flex justify-between text-sm">
            <span className="text-gray-600">{props.taxLabel}</span>
            <span className="text-gray-900">
              {formatAmount(currency, props.tax)}
            </span>
          </div>
          <div className="border-t border-t-gray-900 pt-3 flex justify-between text-lg font-bold text-gray-900">
            <span>Total</span>
            <span>{formatAmount(currency, props.total)}</span>
          </div>
        </div>

        <p className="text-sm text-gray-600 mb-6 break-words whitespace-pre-line">
          {props.note}
        </p>

        <button
          type="button"
          className="w-full bg-indigo-600 hover:bg-indigo-700 text-white font-medium text-sm py-3 px-4 rounded-lg"
        >
          {props.payLabel}
        </button>
      </div>
    </div>
  );
}
