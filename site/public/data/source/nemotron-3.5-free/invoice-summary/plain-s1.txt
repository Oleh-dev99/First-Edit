import { FC } from "react";

interface Props {
  invoiceNumber: string;
  issueDate: string;
  dueDate: string;
  currency: string;
  seller: Party;
  buyer: Party;
  lineItems: LineItem[];
  subtotal: number;
  taxLabel: string;
  tax: number;
  total: number;
  note: string;
  payLabel: string;
}

interface Party {
  name: string;
  address: Address;
}

interface Address {
  street: string;
  city: string;
  postalCode: string;
  country: string;
}

interface LineItem {
  id: string;
  description: string;
  quantity: number;
  unitPrice: number;
  amount: number;
}

const formatMoney = (currency: string, value: number): string => {
  return `${currency} ${value.toLocaleString("en-US", {
    minimumFractionDigits: 2,
    maximumFractionDigits: 2,
  })}`;
};

const Invoice: FC<Props> = ({
  invoiceNumber,
  issueDate,
  dueDate,
  currency,
  seller,
  buyer,
  lineItems,
  subtotal,
  taxLabel,
  tax,
  total,
  note,
  payLabel,
}) => {
  return (
    <div className="flex min-h-screen items-center justify-center bg-gray-50 p-4">
      <div className="max-w-lg w-full bg-white rounded-xl shadow-sm border border-gray-200 p-8">
        <div className="flex flex-col md:flex-row md:items-baseline justify-between gap-4 mb-8">
          <div>
            <h2 className="text-sm font-semibold uppercase tracking-wider text-gray-400">
              Invoice
            </h2>
            <p className="text-base font-semibold text-gray-900">
              Invoice #{invoiceNumber}
            </p>
          </div>
          <div className="text-right">
            <p className="text-xs text-gray-400">Issue date</p>
            <p className="text-sm font-medium text-gray-900">{issueDate}</p>
            <p className="text-xs text-gray-400">Due date</p>
            <p className="text-sm font-medium text-gray-900">{dueDate}</p>
          </div>
        </div>

        <div className="grid grid-cols-2 gap-8 mb-8">
          <div>
            <h3 className="text-xs font-semibold uppercase tracking-wider text-gray-400 mb-2">
              From
            </h3>
            <p className="text-base font-semibold text-gray-900">{seller.name}</p>
            <p className="text-sm text-gray-600">{seller.address.street}</p>
            <p className="text-sm text-gray-600">
              {seller.address.city}, {seller.address.postalCode}
            </p>
            <p className="text-sm text-gray-600">{seller.address.country}</p>
          </div>
          <div>
            <h3 className="text-xs font-semibold uppercase tracking-wider text-gray-400 mb-2">
              Bill to
            </h3>
            <p className="text-base font-semibold text-gray-900">{buyer.name}</p>
            <p className="text-sm text-gray-600">{buyer.address.street}</p>
            <p className="text-sm text-gray-600">
              {buyer.address.city}, {buyer.address.postalCode}
            </p>
            <p className="text-sm text-gray-600">{buyer.address.country}</p>
          </div>
        </div>

        <table className="w-full mb-6 border-collapse text-sm">
          <thead>
            <tr className="border-b border-gray-200">
              <th className="pb-3 text-left text-xs font-semibold text-gray-400 uppercase">
                Description
              </th>
              <th className="pb-3 text-right text-xs font-semibold text-gray-400 uppercase">
                Qty
              </th>
              <th className="pb-3 text-right text-xs font-semibold text-gray-400 uppercase">
                Unit price
              </th>
              <th className="pb-3 text-right text-xs font-semibold text-gray-400 uppercase">
                Amount
              </th>
            </tr>
          </thead>
          <tbody className="divide-y divide-gray-100">
            {lineItems.map((item) => (
              <tr key={item.id} className="text-gray-900">
                <td className="py-3">{item.description}</td>
                <td className="py-3 text-right text-gray-600">{item.quantity}</td>
                <td className="py-3 text-right text-gray-600">
                  {formatMoney(currency, item.unitPrice)}
                </td>
                <td className="py-3 text-right font-medium">
                  {formatMoney(currency, item.amount)}
                </td>
              </tr>
            ))}
          </tbody>
        </table>

        <div className="space-y-2 mb-8">
          <div className="flex justify-between text-sm">
            <span className="text-gray-600">Subtotal</span>
            <span className="text-gray-900">{formatMoney(currency, subtotal)}</span>
          </div>
          <div className="flex justify-between text-sm">
            <span className="text-gray-600">{taxLabel}</span>
            <span className="text-gray-900">{formatMoney(currency, tax)}</span>
          </div>
          <div className="flex justify-between mt-2 pt-2 text-base font-bold border-t border-gray-200">
            <span className="text-gray-900">Total</span>
            <span className="text-gray-900">{formatMoney(currency, total)}</span>
          </div>
        </div>

        <p className="text-sm text-gray-600 mb-6">{note}</p>

        <button
          type="button"
          className="w-full py-3.5 px-4 bg-blue-600 text-white font-semibold rounded-lg hover:bg-blue-700 focus:outline-none focus:ring-2 focus:ring-blue-500 focus:ring-offset-2"
        >
          {payLabel}
        </button>
      </div>
    </div>
  );
};

export default Invoice;
