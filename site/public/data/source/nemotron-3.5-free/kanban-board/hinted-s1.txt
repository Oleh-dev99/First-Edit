export interface Props {
  columns: Column[];
}

export interface Column {
  id: string;
  title: string;
  cards: Card[];
}

export interface Card {
  id: string;
  title: string;
  tags: string[];
  assignee: string;
  dueDate: string;
  commentCount: number;
}

function userInitials(name: string): string {
  return (
    name
      .split(/\s+/)
      .filter((word) => word.length > 0)
      .map((word) => word.charAt(0).toUpperCase())
      .slice(0, 2)
      .join("") || "?"
  );
}

export default function ProjectBoard({ columns }: Props) {
  return (
    <section className="w-full px-4 py-6">
      <div className="flex flex-col lg:flex-row gap-4 overflow-x-auto min-w-0">
        {columns.map((column) => (
          <div
            key={column.id}
            className="flex flex-col gap-3 w-full min-w-[260px] lg:flex-none lg:w-[340px]"
          >
            <div className="flex flex-wrap items-center justify-between gap-1.5 min-w-0">
              <h2 className="text-sm font-semibold text-gray-900 break-words">
                {column.title}
              </h2>
              <span className="text-xs font-medium text-gray-500 bg-gray-100 rounded-full px-2 py-0.5">
                {column.cards.length} {column.cards.length === 1 ? "card" : "cards"}
              </span>
            </div>
            <div className="flex flex-col gap-2">
              {column.cards.map((card) => (
                <div
                  key={card.id}
                  className="flex flex-col gap-2 rounded-lg border border-gray-200 bg-white p-3 shadow-sm min-w-0"
                >
                  <h3 className="text-sm font-semibold text-gray-900 leading-snug break-words">
                    {card.title}
                  </h3>
                  <div className="flex flex-wrap gap-1.5">
                    {card.tags.map((tag, i) => (
                      <span
                        key={`${i}-${tag}`}
                        className="text-xs rounded bg-blue-50 px-2 py-0.5 border border-blue-100 text-blue-700 break-words"
                      >
                        {tag}
                      </span>
                    ))}
                  </div>
                  <div className="flex flex-wrap items-center gap-3 text-xs text-gray-500 min-w-0">
                    <div className="flex flex-wrap items-center gap-1.5 min-w-0">
                      <div className="flex-shrink-0 h-6 w-6 rounded-full bg-gray-200 flex items-center justify-center text-[10px] font-medium text-gray-600">
                        {userInitials(card.assignee)}
                      </div>
                      <span className="text-gray-600 break-words">{card.assignee}</span>
                    </div>
                    <span className="break-words">{card.dueDate}</span>
                    <span className="break-words">
                      {card.commentCount} {card.commentCount === 1 ? "comment" : "comments"}
                    </span>
                  </div>
                </div>
              ))}
            </div>
          </div>
        ))}
      </div>
    </section>
  );
}
