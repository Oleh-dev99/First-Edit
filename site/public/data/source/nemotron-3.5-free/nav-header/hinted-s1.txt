import * as React from "react";

export interface Props {
  productName: string;
  links: NavLink[];
  ctaLabel: string;
  user: User;
}

export interface NavLink {
  label: string;
  href: string;
  active: boolean;
}

export interface User {
  name: string;
  email: string;
}

function userInitials(user: User): string {
  const parts = user.name.trim().split(/\s+/).filter(Boolean);
  if (parts.length === 0) return "?";
  return parts
    .map((part) => part.charAt(0).toUpperCase())
    .join("")
    .slice(0, 2);
}

export default function AppHeader({
  productName,
  links,
  ctaLabel,
  user,
}: Props) {
  const initials = userInitials(user);

  return (
    <header className="sticky top-0 z-40 border-b border-slate-200 bg-white">
      <div className="mx-auto max-w-7xl px-4 sm:px-6 lg:px-8">
        <div className="flex flex-wrap items-center justify-between gap-3">
          <div className="flex items-center gap-2 min-w-0">
            <span className="text-xl font-semibold tracking-tight text-slate-900 break-words">
              {productName}
            </span>
          </div>

          <nav className="flex flex-wrap gap-2 items-center min-w-0">
            {links.map((link) => (
              <a
                key={link.href}
                href={link.href}
                className={
                  link.active
                    ? "bg-indigo-600 text-white text-sm font-medium px-3 py-2 rounded-md break-words"
                    : "text-slate-600 text-sm px-3 py-2 rounded-md hover:bg-slate-100 hover:text-indigo-600 break-words"
                }
              >
                {link.label}
              </a>
            ))}
          </nav>

          <div className="flex flex-wrap items-center gap-3 min-w-0">
            <button
              type="button"
              className="bg-indigo-600 text-white text-sm font-medium px-4 py-2 rounded-md whitespace-normal"
            >
              {ctaLabel}
            </button>
            <div className="flex flex-wrap items-center gap-2 min-w-0">
              <div className="flex flex-col gap-1 min-w-0">
                <span className="text-sm font-medium text-slate-900 break-words">
                  {user.name}
                </span>
                <span className="text-xs text-slate-500 break-words">
                  {user.email}
                </span>
              </div>
              <div
                className="w-8 h-8 shrink-0 rounded-full bg-indigo-100 flex items-center justify-center text-sm font-medium text-indigo-700"
              >
                {initials}
              </div>
            </div>
          </div>
        </div>
      </div>
    </header>
  );
}
