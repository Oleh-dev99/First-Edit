import { type FC } from "react";

export interface Props {
  productName: string;
  links: NavLink[];
  ctaLabel: string;
  user: User;
}

export interface NavLink {
  label: string;
  href: string;
  active: boolean;
}

export interface User {
  name: string;
  email: string;
}

const LedgerloomHeader: FC<Props> = (props) => {
  const { productName, links, ctaLabel, user } = props;

  const initials = user.name
    .split(/\s+/)
    .filter(Boolean)
    .map((part) => part.charAt(0).toUpperCase())
    .slice(0, 2)
    .join("") || "?";

  return (
    <header className="w-full border-b border-gray-200 bg-white">
      <div className="mx-auto flex max-w-7xl flex-col sm:flex-row sm:items-center sm:justify-between gap-4 px-4 py-3 sm:px-6 lg:px-8">
        <div className="flex-shrink-0">
          <span
            className="text-lg font-bold text-gray-900"
            style={{ overflowWrap: "break-word" }}
          >
            {productName}
          </span>
        </div>

        <nav className="flex flex-col sm:flex-row sm:flex-wrap gap-2 sm:gap-4 min-w-0 w-full sm:w-auto">
          {links.map((link, index) => (
            <a
              key={index}
              href={link.href}
              className={`px-3 py-2 text-sm font-medium rounded-md transition-colors flex items-center gap-1 ${
                link.active
                  ? "bg-blue-600 text-white"
                  : "text-gray-600 hover:bg-gray-100 hover:text-gray-900"
              }`}
              style={{ overflowWrap: "break-word" }}
            >
              {link.label}
            </a>
          ))}
        </nav>

        <div className="flex flex-col sm:flex-row items-center justify-between gap-3 w-full sm:w-auto">
          <button
            className="inline-flex w-full sm:w-auto items-center justify-center gap-2 px-4 py-2 text-sm font-medium text-white bg-blue-600 rounded-md shadow-sm hover:bg-blue-700"
            type="button"
            style={{ overflowWrap: "break-word" }}
          >
            {ctaLabel}
          </button>

          <div className="flex items-center gap-3 min-w-0">
            <div className="flex-shrink-0 w-8 h-8 rounded-full bg-gray-200 flex items-center justify-center text-sm font-medium text-gray-600">
              {initials}
            </div>
            <div className="min-w-0 flex flex-col">
              <span
                className="text-sm font-medium text-gray-900 leading-tight"
                style={{ overflowWrap: "break-word" }}
              >
                {user.name}
              </span>
              <span
                className="text-xs text-gray-500 leading-tight"
                style={{ overflowWrap: "break-word" }}
              >
                {user.email}
              </span>
            </div>
          </div>
        </div>
      </div>
    </header>
  );
};

export default LedgerloomHeader;
