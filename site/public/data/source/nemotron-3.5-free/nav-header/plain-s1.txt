import { type FunctionComponent } from 'react';

export interface NavLink {
  label: string;
  href: string;
  active: boolean;
}

export interface User {
  name: string;
  email: string;
}

export interface Props {
  productName: string;
  links: NavLink[];
  ctaLabel: string;
  user: User;
}

const userInitials = (name: string): string => {
  return name
    .trim()
    .split(/\s+/)
    .filter((part) => part.length > 0)
    .map((part) => part.charAt(0))
    .join('')
    .toUpperCase()
    .slice(0, 2) || '?';
};

const AppHeader: FunctionComponent<Props> = ({
  productName,
  links,
  ctaLabel,
  user,
}) => {
  const initials = userInitials(user.name);

  return (
    <header className="sticky top-0 z-40 w-full border-b border-slate-200 bg-white">
      <div className="flex items-center justify-between px-4 py-3 sm:px-6 lg:px-8">
        <div className="flex items-center gap-4">
          <span className="text-xl font-semibold text-slate-900 tracking-tight">
            {productName}
          </span>
          <nav className="flex items-center gap-1 hidden sm:flex">
            {links.map((link) => (
              <a
                key={link.href}
                href={link.href}
                className={
                  link.active
                    ? 'rounded-md text-sm font-medium px-3 py-1.5 bg-slate-100 text-slate-900'
                    : 'rounded-md text-sm font-medium px-3 py-1.5 text-slate-500 hover:text-slate-900 hover:bg-slate-50'
                }
              >
                {link.label}
              </a>
            ))}
          </nav>
        </div>
        <div className="flex items-center gap-3">
          <button className="flex items-center gap-2 rounded-md bg-slate-900 px-4 py-2 text-sm font-semibold text-white shadow-sm hover:bg-slate-800 transition-colors">
            {ctaLabel}
          </button>
          <div className="hidden sm:flex flex-col items-end gap-1 leading-none">
            <span className="text-sm font-medium text-slate-900">{user.name}</span>
            <span className="text-xs text-slate-500">{user.email}</span>
          </div>
          <div className="flex-shrink-0 h-9 w-9 rounded-full bg-slate-100 flex items-center justify-center text-sm font-semibold text-slate-600">
            {initials}
          </div>
        </div>
      </div>
    </header>
  );
};

export default AppHeader;
