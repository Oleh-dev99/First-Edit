import * as React from "react";

export interface ConfirmInput {
  label: string;
  placeholder: string;
}

export interface Props {
  title: string;
  body: string;
  projectName: string;
  consequencesTitle: string;
  consequences: string[];
  confirmInput: ConfirmInput;
  cancelLabel: string;
  confirmLabel: string;
}

export default function DeleteProjectDialog(props: Props): React.JSX.Element {
  const { title, body, projectName, consequencesTitle, consequences, confirmInput, cancelLabel, confirmLabel } = props;

  return (
    <div
      role="dialog"
      aria-modal="true"
      aria-labelledby="delete-project-title"
      aria-describedby="delete-project-body"
      className="bg-white rounded-xl border border-gray-200 shadow-lg p-6 w-full max-w-[90vw] text-sm"
    >
      <h2 id="delete-project-title" className="text-lg font-semibold text-gray-900">
        {title}
      </h2>
      <p id="delete-project-body" className="mt-2 text-gray-600 leading-relaxed">
        {body}
      </p>

      <div className="mt-4 rounded border border-gray-200 bg-gray-50 p-3">
        <p className="font-mono text-sm text-gray-900 break-all">{projectName}</p>
      </div>

      {consequences.length > 0 && (
        <>
          <h3 className="mt-4 mb-2 text-sm font-semibold text-gray-900">{consequencesTitle}</h3>
          <ul className="list-disc list-inside space-y-1 text-gray-600 mb-4">
            {consequences.map((item, index) => (
              <li key={index}>{item}</li>
            ))}
          </ul>
        </>
      )}

      <div className="mb-4">
        <label htmlFor="confirm-project-name" className="block text-sm font-medium text-gray-700 mb-1">
          {confirmInput.label}
        </label>
        <input
          id="confirm-project-name"
          type="text"
          placeholder={confirmInput.placeholder}
          className="w-full rounded border border-gray-300 bg-white px-3 py-2 text-sm"
        />
      </div>

      <div className="flex flex-col sm:flex-row gap-2 justify-end">
        <button type="button" className="px-4 py-2 rounded border border-gray-300 bg-white text-gray-700 font-medium hover:bg-gray-50">
          {cancelLabel}
        </button>
        <button type="button" className="px-4 py-2 rounded bg-red-600 text-white font-medium hover:bg-red-700">
          {confirmLabel}
        </button>
      </div>
    </div>
  );
}
