export interface Contact {
  name: string;
  status: string;
}

export type Sender = 'me' | 'them';

export interface Message {
  id: string;
  from: Sender;
  author: string;
  text: string;
  time: string;
}

export interface Composer {
  placeholder: string;
  sendLabel: string;
}

export interface Props {
  contact: Contact;
  messages: Message[];
  composer: Composer;
}

function initials(name: string): string {
  return name
    .split(/\s+/)
    .filter(Boolean)
    .map((part) => part.charAt(0).toUpperCase())
    .join('')
    .slice(0, 2);
}

export default function ChatThread({ contact, messages, composer }: Props) {
  return (
    <div className="flex flex-col border border-gray-200 rounded-xl bg-white shadow-sm w-full max-w-md">
      <div className="flex items-center gap-3 p-3 border-b border-gray-100">
        <div className="flex-shrink-0 w-10 h-10 rounded-full bg-gray-200 flex items-center justify-center text-sm font-semibold text-gray-700">
          {initials(contact.name)}
        </div>
        <div className="flex-1 min-w-0">
          <div className="text-sm font-semibold text-gray-900 leading-tight">
            {contact.name}
          </div>
          <div className="flex items-center gap-1.5 text-xs text-gray-500">
            <span className="w-1.5 h-1.5 rounded-full bg-green-500"></span>
            {contact.status}
          </div>
        </div>
      </div>

      <div className="flex-1 overflow-y-auto p-3 space-y-3">
        {messages.map((message) => (
          <div key={message.id} className="flex w-full flex-col">
            <div
              className={`max-w-[80%] rounded-lg px-3 py-2 text-sm leading-relaxed ${
                message.from === 'me'
                  ? 'bg-blue-600 text-white ml-auto text-right'
                  : 'bg-gray-100 text-gray-900 mr-auto text-left'
              }`}
            >
              <div className={`text-xs mb-0.5 font-medium ${
                message.from === 'me' ? 'text-blue-100' : 'text-gray-600'
              }`}>
                {message.author}
              </div>
              <div className="whitespace-pre-wrap">{message.text}</div>
              <div className={`text-[10px] mt-0.5 ${
                message.from === 'me' ? 'text-blue-200/80' : 'text-gray-400'
              }`}>
                {message.time}
              </div>
            </div>
          </div>
        ))}
      </div>

      <div className="p-3 border-t border-gray-100 flex gap-2 bg-white/50">
        <input
          type="text"
          placeholder={composer.placeholder}
          className="flex-1 min-w-0 rounded-lg border border-gray-300 bg-white px-3 py-2 text-sm text-gray-900 placeholder-gray-500 focus:outline-none focus:ring-2 focus:ring-blue-500 focus:border-transparent"
        />
        <button type="button" className="flex-shrink-0 bg-blue-600 text-white text-sm font-medium px-3 py-2 rounded-lg hover:bg-blue-700">
          {composer.sendLabel}
        </button>
      </div>
    </div>
  );
}
