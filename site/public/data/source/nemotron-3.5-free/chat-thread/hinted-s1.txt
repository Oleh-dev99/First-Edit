import { type FC } from "react";

interface Contact {
  name: string;
  status: string;
}

type Sender = "me" | "them";

interface Message {
  id: string;
  from: Sender;
  author: string;
  text: string;
  time: string;
}

interface Composer {
  placeholder: string;
  sendLabel: string;
}

interface Props {
  contact: Contact;
  messages: Message[];
  composer: Composer;
}

function getInitials(name: string): string {
  return name
    .trim()
    .split(/\s+/)
    .filter(Boolean)
    .slice(0, 2)
    .map((word) => word.charAt(0).toUpperCase())
    .join("") || "?";
}

const ChatThread: FC<Props> = ({ contact, messages, composer }) => {
  return (
    <div className="w-full max-w-xs flex flex-col border border-slate-200 rounded-lg bg-white shadow-sm">
      <header className="flex items-center gap-2 p-3 border-b border-slate-200 bg-white">
        <div className="w-8 h-8 rounded-full bg-slate-200 flex items-center justify-center text-slate-600 text-sm font-medium">
          {getInitials(contact.name)}
        </div>
        <div className="flex-1 min-w-0">
          <div className="text-sm font-semibold text-slate-900 leading-tight">
            {contact.name}
          </div>
          <div className="text-xs text-slate-500">{contact.status}</div>
        </div>
      </header>
      <main className="flex-1 overflow-y-auto p-3 space-y-3">
        {messages.length === 0 ? (
          <p className="text-xs text-slate-400 text-center">No messages yet</p>
        ) : (
          messages.map((message) => {
            const isMe = message.from === "me";
            return (
              <div
                key={message.id}
                className={`flex w-full ${isMe ? "justify-end" : "justify-start"}`}
              >
                <div
                  className={`max-w-[85%] break-words whitespace-pre-wrap ${isMe ? "text-right" : "text-left"}`}
                >
                  <div className="text-xs font-medium mb-0.5">
                    {message.author}
                  </div>
                  <div
                    className={`inline-block rounded-lg p-2 text-sm ${isMe ? "bg-blue-50 border border-blue-200 text-slate-900" : "bg-slate-100 border border-slate-200 text-slate-900"}`}
                  >
                    {message.text}
                  </div>
                  <div className="text-[10px] text-slate-400 mt-0.5">
                    {message.time}
                  </div>
                </div>
              </div>
            );
          })
        )}
      </main>
      <footer className="p-3 border-t border-slate-200 bg-white flex gap-2">
        <textarea
          className="flex-1 min-w-0 min-h-[80px] resize-none border border-slate-300 rounded-lg p-2 text-sm text-slate-900 placeholder-slate-400 focus:outline-none focus:ring-2 focus:ring-blue-500"
          placeholder={composer.placeholder}
        />
        <button
          className="flex-shrink-0 bg-slate-900 text-white px-3 py-2 rounded-lg text-sm font-medium hover:bg-slate-800 disabled:opacity-50"
        >
          {composer.sendLabel}
        </button>
      </footer>
    </div>
  );
};

export default ChatThread;
