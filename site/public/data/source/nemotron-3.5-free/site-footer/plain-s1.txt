import { type FC } from "react";

export interface Props {
  companyName: string;
  tagline: string;
  columns: LinkColumn[];
  newsletter: Newsletter;
  legal: string;
}

export interface LinkColumn {
  title: string;
  links: FooterLink[];
}

export interface FooterLink {
  label: string;
  href: string;
}

export interface Newsletter {
  prompt: string;
  label: string;
  placeholder: string;
  buttonLabel: string;
}

const Footer: FC<Props> = ({
  companyName,
  tagline,
  columns,
  newsletter,
  legal,
}) => {
  return (
    <footer className="bg-slate-950 text-slate-300">
      <div className="mx-auto max-w-7xl px-6 py-12 sm:px-8 lg:px-12">
        <div className="grid grid-cols-1 gap-12 md:grid-cols-12 lg:gap-8">
          <div className="md:col-span-4 lg:col-span-3">
            <h2 className="text-xl font-bold text-white">{companyName}</h2>
            <p className="mt-4 text-sm text-slate-400 leading-relaxed">
              {tagline}
            </p>
          </div>

          {columns.map((column, columnIndex) => (
            <div key={columnIndex}>
              <h3 className="text-xs font-semibold text-white uppercase tracking-wider mb-4">
                {column.title}
              </h3>
              <ul className="space-y-2">
                {column.links.map((link, linkIndex) => (
                  <li key={linkIndex}>
                    <a
                      href={link.href}
                      className="text-sm text-slate-400 hover:text-white transition-colors"
                    >
                      {link.label}
                    </a>
                  </li>
                ))}
              </ul>
            </div>
          ))}

          <div className="md:col-span-4 lg:col-span-3">
            <p className="text-sm text-slate-400 mb-4">{newsletter.prompt}</p>
            <form className="flex flex-col sm:flex-row gap-3" noValidate>
              <div className="flex-1 min-w-0">
                <label
                  htmlFor="footer-newsletter-email"
                  className="block text-xs font-medium text-slate-400 mb-1"
                >
                  {newsletter.label}
                </label>
                <input
                  id="footer-newsletter-email"
                  type="email"
                  placeholder={newsletter.placeholder}
                  className="flex-1 min-w-0 px-3 py-2 text-sm text-slate-300 bg-slate-900 border border-slate-800 rounded-md focus:outline-none focus:ring-2 focus:ring-slate-500 focus:border-slate-500"
                />
              </div>
              <button
                type="submit"
                className="flex-shrink-0 px-4 py-2 text-sm font-semibold text-white bg-slate-700 hover:bg-slate-600 rounded-md"
              >
                {newsletter.buttonLabel}
              </button>
            </form>
          </div>
        </div>

        <div className="mt-12 pt-8 border-t border-slate-800">
          <p className="text-xs text-slate-500">{legal}</p>
        </div>
      </div>
    </footer>
  );
};

export default Footer;
