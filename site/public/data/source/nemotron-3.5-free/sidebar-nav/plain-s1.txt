export interface Props {
  workspaceName: string;
  sections: NavSection[];
  user: SidebarUser;
}

export interface NavSection {
  title: string;
  links: NavLink[];
}

export interface NavLink {
  label: string;
  href: string;
  count: number | null;
  active: boolean;
}

export interface SidebarUser {
  name: string;
  role: string;
}

export default function Sidebar({ workspaceName, sections, user }: Props) {
  return (
    <aside className="fixed inset-y-0 left-0 z-50 flex flex-col gap-6 bg-white/90 backdrop-blur-md border-r border-gray-200 px-4 py-6 text-sm">
      <div className="flex items-center gap-2 mb-1">
        <span className="text-lg font-semibold text-gray-900">{workspaceName}</span>
      </div>

      {sections.map((section) => (
        <div key={section.title}>
          <h2 className="mx-2 mb-1 text-xs font-semibold tracking-wider text-gray-500 uppercase">
            {section.title}
          </h2>
          <nav className="space-y-0.5">
            {section.links.map((link) => (
              <a
                key={link.href}
                href={link.href}
                className={
                  link.active
                    ? "flex items-center gap-2 px-3 py-2 text-sm font-medium rounded-md bg-indigo-600 text-white"
                    : "flex items-center gap-2 px-3 py-2 text-sm rounded-md text-gray-700 hover:bg-gray-100 hover:text-gray-900"
                }
              >
                {link.label}
                {link.count !== null && (
                  <span
                    className={`ml-auto inline-flex items-center justify-center min-w-[18px] h-6 px-1.5 rounded-full text-xs font-medium ${link.active ? "bg-white/20 text-white" : "bg-gray-100 text-gray-600"}`}
                  >
                    {link.count}
                  </span>
                )}
              </a>
            ))}
          </nav>
        </div>
      ))}

      <div className="mt-auto pt-6 border-t border-gray-200 flex flex-col gap-1">
        <p className="text-sm font-medium text-gray-900">{user.name}</p>
        <p className="text-xs text-gray-500">{user.role}</p>
      </div>
    </aside>
  );
}
