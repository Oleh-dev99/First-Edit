import * as React from "react";

export interface ConfirmInput {
  label: string;
  placeholder: string;
}

/** A dialog that confirms the deletion of a project. */
export interface Props {
  title: string;
  /** Explains what the action does. */
  body: string;
  /** The name of the project that will be deleted. */
  projectName: string;
  /** Heading of the list of consequences. */
  consequencesTitle: string;
  /** What deleting the project will do. */
  consequences: string[];
  /** The input where the user types the project's name to confirm. */
  confirmInput: ConfirmInput;
  cancelLabel: string;
  /** Label of the button that deletes the project. */
  confirmLabel: string;
}

const DeleteProjectDialog = (props: Props) => {
  const titleId = React.useId();
  const descriptionId = React.useId();
  const inputId = React.useId();

  return (
    <div
      role="dialog"
      aria-labelledby={titleId}
      aria-describedby={descriptionId}
      className="mx-auto w-full max-w-[560px] rounded-2xl border border-slate-200 bg-white text-slate-900 shadow-2xl"
    >
      <div className="flex items-start gap-4 px-6 pt-6">
        <div
          aria-hidden="true"
          className="flex h-10 w-10 shrink-0 items-center justify-center rounded-full bg-red-50 text-lg font-bold text-red-600 ring-1 ring-red-100"
        >
          !
        </div>
        <div className="min-w-0">
          <h2
            id={titleId}
            className="text-lg font-semibold tracking-tight text-slate-900"
          >
            {props.title}
          </h2>
          <p
            id={descriptionId}
            className="mt-1.5 text-sm leading-6 text-slate-600"
          >
            {props.body}
          </p>
        </div>
      </div>

      <div className="px-6 pt-5">
        <div className="rounded-xl border border-slate-200 bg-slate-50 px-4 py-3.5">
          <p className="text-[11px] font-semibold uppercase tracking-[0.14em] text-slate-500">
            Project to delete
          </p>
          <p className="mt-1.5 break-all font-mono text-sm font-semibold text-slate-900">
            {props.projectName}
          </p>
        </div>
      </div>

      <div className="px-6 pt-5">
        <h3 className="text-sm font-semibold text-slate-900">
          {props.consequencesTitle}
        </h3>
        <ul className="mt-3 space-y-2.5">
          {props.consequences.map((consequence, index) => (
            <li
              key={`${index}-${consequence}`}
              className="flex gap-3 text-sm leading-6 text-slate-600"
            >
              <span
                aria-hidden="true"
                className="mt-[9px] h-1.5 w-1.5 shrink-0 rounded-full bg-red-500"
              />
              <span className="min-w-0">{consequence}</span>
            </li>
          ))}
        </ul>
      </div>

      <div className="px-6 pt-5">
        <label
          htmlFor={inputId}
          className="block text-sm font-medium text-slate-800"
        >
          {props.confirmInput.label}
        </label>
        <input
          id={inputId}
          type="text"
          placeholder={props.confirmInput.placeholder}
          autoComplete="off"
          spellCheck={false}
          className="mt-2 block w-full rounded-lg border border-slate-300 bg-white px-3.5 py-2.5 text-sm text-slate-900 shadow-sm outline-none transition placeholder:text-slate-400 focus:border-red-500 focus:ring-4 focus:ring-red-100"
        />
      </div>

      <div className="mt-6 flex flex-col gap-2.5 border-t border-slate-100 bg-slate-50/60 px-6 py-5 sm:flex-row sm:justify-end">
        <button
          type="button"
          className="inline-flex h-10 w-full items-center justify-center rounded-lg border border-slate-300 bg-white px-4 py-2.5 text-sm font-semibold text-slate-700 shadow-sm transition hover:bg-slate-50 focus:outline-none focus-visible:ring-4 focus-visible:ring-slate-200 sm:w-auto"
        >
          {props.cancelLabel}
        </button>
        <button
          type="button"
          className="inline-flex h-10 w-full items-center justify-center rounded-lg bg-red-600 px-4 py-2.5 text-sm font-semibold text-white shadow-sm transition hover:bg-red-700 focus:outline-none focus-visible:ring-4 focus-visible:ring-red-200 sm:w-auto"
        >
          {props.confirmLabel}
        </button>
      </div>
    </div>
  );
};

export default DeleteProjectDialog;
