interface ConfirmInput {
  label: string;
 placeholder: string;
}

/** A dialog that confirms the deletion of a project. */
export interface Props {
  title: string;
  /** Explains what the action does. */
  body: string;
  /** The name of the project that will be deleted. */
  projectName: string;
  /** Heading of the list of consequences. */
  consequencesTitle: string;
  /** What deleting the project will do. */
  consequences: string[];
  /** The input where the user types the project's name to confirm. */
  confirmInput: ConfirmInput;
  cancelLabel: string;
  /** Label of the button that deletes the project. */
  confirmLabel: string;
}

export default function DeleteProjectDialog({
  title,
  body,
  projectName,
  consequencesTitle,
  consequences,
  confirmInput,
  cancelLabel,
  confirmLabel,
}: Props) {
  return (
    <section
      aria-labelledby="delete-project-title"
      className="w-full min-w-0 rounded-2xl border border-slate-200 bg-white p-5 text-slate-900 shadow-xl sm:p-6"
    >
      <div className="flex gap-3 sm:gap-4">
        <span
          aria-hidden="true"
          className="flex size-10 shrink-0 items-center justify-center rounded-full bg-red-50 text-red-600 ring-1 ring-red-100"
        >
          <svg
            viewBox="0 0 24 24"
            fill="none"
            stroke="currentColor"
            strokeWidth="1.8"
            strokeLinecap="round"
            strokeLinejoin="round"
            className="size-5"
          >
            <path d="M4 7h16" />
            <path d="M9 7V5.5A1.5 1.5 0 0 1 10.5 4h3A1.5 1.5 0 0 1 15 5.5V7" />
            <path d="M6.5 7l.8 11a2 2 0 0 0 2 1.9h5.4a2 2 0 0 0 2-1.9l.8-11" />
            <path d="M10.5 11v5.5M13.5 11v5.5" />
          </svg>
        </span>

        <div className="min-w-0 flex-1 space-y-4">
          <div className="space-y-1.5">
            <h2
              id="delete-project-title"
              className="text-lg font-semibold tracking-tight text-slate-900 sm:text-xl"
            >
              {title}
            </h2>
            <p className="text-sm leading-6 text-slate-600">{body}</p>
          </div>

          <p className="text-xs font-medium uppercase tracking-wide text-slate-500">
            Project
          </p>
          <div className="rounded-xl border border-slate-200 bg-slate-50 px-4 py-3">
            <p className="font-mono text-sm font-medium text-slate-900">
              {projectName}
            </p>
          </div>

          {consequences.length > 0 && (
            <div className="space-y-2">
              <h3 className="text-sm font-semibold text-slate-900">
                {consequencesTitle}
              </h3>
              <ul className="space-y-2 pl-5 text-sm leading-6 text-slate-600 marker:text-slate-400">
                {consequences.map((consequence, index) => (
                  <li key={`${index}-${consequence}`}>{consequence}</li>
                ))}
              </ul>
            </div>
          )}

          <div className="space-y-1.5">
            <label
              htmlFor="delete-project-confirm"
              className="block text-sm font-medium text-slate-900"
            >
              {confirmInput.label}
            </label>
            <input
              id="delete-project-confirm"
              type="text"
              placeholder={confirmInput.placeholder}
              autoComplete="off"
              spellCheck={false}
              className="block w-full min-w-0 rounded-lg border border-slate-300 bg-white px-3 py-2 text-sm text-slate-900 shadow-sm placeholder:text-slate-400 focus:border-red-400 focus:outline-none focus:ring-2 focus:ring-red-500/30"
            />
          </div>

          <div className="flex flex-col-reverse gap-2 pt-1 sm:flex-row sm:justify-end">
            <button
              type="button"
              className="inline-flex w-full min-w-0 items-center justify-center rounded-lg border border-slate-300 bg-white px-4 py-2 text-sm font-semibold text-slate-700 transition-colors hover:bg-slate-50 focus:outline-none focus:ring-2 focus:ring-slate-400 focus:ring-offset-2 sm:w-auto"
            >
              {cancelLabel}
            </button>
            <button
              type="button"
              className="inline-flex w-full min-w-0 items-center justify-center rounded-lg bg-red-600 px-4 py-2 text-sm font-semibold text-white shadow-sm transition-colors hover:bg-red-700 focus:outline-none focus:ring-2 focus:ring-red-500 focus:ring-offset-2 sm:w-auto"
            >
              {confirmLabel}
            </button>
          </div>
        </div>
      </div>
    </section>
  );
}
