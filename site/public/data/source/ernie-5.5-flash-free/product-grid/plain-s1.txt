import React from "react";

interface Props {
  heading: string;
  /** e.g. "Showing 6 of 48 products" */
  resultsSummary: string;
  /** ISO 4217 code used for every price, e.g. "USD". */
  currency: string;
  products: Product[];
}

interface Product {
  id: string;
  name: string;
  image: ProductImage;
  price: number;
  /** The price before the sale, present when the product is on sale. */
  originalPrice?: number;
  /** Average rating from 0 to 5. */
  rating: number;
  reviewCount: number;
  tags: string[];
  /** Label of the add-to-cart button. */
  ctaLabel: string;
}

interface ProductImage {
  /** Path of the image, e.g. "/fe-assets/placeholder-1x1.svg". */
  src: string;
  /** Text alternative for the image. */
  alt: string;
}

const formatPrice = (value: number, currency: string) =>
  new Intl.NumberFormat("en-US", {
    style: "currency",
    currency,
  }).format(value);

const Star = () => (
  <svg
    viewBox="0 0 20 20"
    fill="currentColor"
    className="h-4 w-4 shrink-0"
    aria-hidden="true"
  >
    <path d="M10 1.6l2.47 5.01 5.53.8-4 3.9.94 5.5L10 14.2l-4.94 2.6.94-5.5-4-3.9 5.53-.8L10 1.6z" />
  </svg>
);

const Stars: React.FC<{ rating: number }> = ({ rating }) => {
  const clamped = Math.max(0, Math.min(5, rating));

  return (
    <span
      className="relative inline-flex h-4 w-[5.5rem] align-middle"
      role="img"
      aria-label={`${clamped.toFixed(1)} out of 5 stars`}
    >
      <span className="flex h-full items-center gap-0.5 text-stone-300">
        {[0, 1, 2, 3, 4].map((star) => (
          <Star key={star} />
        ))}
      </span>
      <span
        className="absolute inset-y-0 left-0 flex h-full items-center gap-0.5 overflow-hidden text-amber-500"
        style={{ width: `${(clamped / 5) * 100}%` }}
      >
        {[0, 1, 2, 3, 4].map((star) => (
          <Star key={star} />
        ))}
      </span>
    </span>
  );
};

export default function ProductListing({
  heading,
  resultsSummary,
  currency,
  products,
}: Props) {
  return (
    <section
      aria-labelledby="product-listing-heading"
      className="w-full bg-[#fbf7f0] px-4 py-10 text-stone-900 sm:px-6 lg:px-10 lg:py-14"
    >
      <div className="mx-auto w-full max-w-[1600px]">
        <header className="flex flex-col gap-3 border-b border-stone-200 pb-6 sm:flex-row sm:items-end sm:justify-between">
          <div>
            <h1
              id="product-listing-heading"
              className="text-3xl font-semibold tracking-tight text-stone-950 sm:text-4xl"
            >
              {heading}
            </h1>
            <p className="mt-2 text-sm text-stone-600">{resultsSummary}</p>
          </div>
        </header>

        <div className="mt-8 grid grid-cols-1 gap-5 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4">
          {products.map((product) => {
            const salePrice =
              product.originalPrice !== undefined &&
              product.originalPrice > product.price
                ? product.originalPrice
                : null;

            return (
              <article
                key={product.id}
                className="group flex h-full flex-col overflow-hidden rounded-2xl border border-stone-200 bg-white shadow-sm transition-shadow hover:shadow-md"
              >
                <div className="relative aspect-[4/3] overflow-hidden bg-stone-100">
                  <img
                    src={product.image.src}
                    alt={product.image.alt}
                    className="h-full w-full object-cover transition-transform duration-300 group-hover:scale-[1.02]"
                    loading="lazy"
                  />
                  {salePrice !== null ? (
                    <span className="absolute left-3 top-3 rounded-full bg-amber-600 px-2.5 py-1 text-xs font-semibold text-white shadow-sm">
                      Sale
                    </span>
                  ) : null}
                </div>

                <div className="flex flex-1 flex-col p-4 sm:p-5">
                  <h2 className="text-base font-semibold leading-snug text-stone-900">
                    {product.name}
                  </h2>

                  <div className="mt-2 flex flex-wrap items-center gap-x-2 gap-y-1">
                    <Stars rating={product.rating} />
                    <span className="text-sm font-medium text-stone-700">
                      {product.rating.toFixed(1)}
                    </span>
                    <span className="text-sm text-stone-500">
                      ({product.reviewCount}{" "}
                      {product.reviewCount === 1 ? "review" : "reviews"})
                    </span>
                  </div>

                  <div className="mt-3 flex flex-wrap gap-1.5">
                    {product.tags.map((tag) => (
                      <span
                        key={`${product.id}-${tag}`}
                        className="rounded-full border border-stone-200 bg-stone-50 px-2.5 py-1 text-xs font-medium text-stone-600"
                      >
                        {tag}
                      </span>
                    ))}
                  </div>

                  <div className="mt-auto pt-5">
                    <div className="flex flex-wrap items-baseline gap-x-2 gap-y-1">
                      <span className="text-xl font-semibold text-stone-950">
                        {formatPrice(product.price, currency)}
                      </span>
                      {salePrice !== null ? (
                        <span className="text-sm text-stone-500 line-through">
                          {formatPrice(salePrice, currency)}
                        </span>
                      ) : null}
                    </div>

                    <button
                      type="button"
                      className="mt-3 w-full rounded-xl bg-stone-900 px-4 py-2.5 text-sm font-semibold text-white transition-colors hover:bg-stone-800 focus:outline-none focus-visible:ring-2 focus-visible:ring-amber-500 focus-visible:ring-offset-2"
                    >
                      {product.ctaLabel}
                    </button>
                  </div>
                </div>
              </article>
            );
          })}
        </div>
      </div>
    </section>
  );
}
