import React, { type CSSProperties } from "react";

export interface Props {
  heading: string;
  /** e.g. "Showing 6 of 48 products" */
  resultsSummary: string;
  /** ISO 4217 code used for every price, e.g. "USD". */
  currency: string;
  products: Product[];
}

export interface Product {
  id: string;
  name: string;
  image: ProductImage;
  price: number;
  /** The price before the sale, present when the product is on sale. */
  originalPrice?: number;
  /** Average rating from 0 to 5. */
  rating: number;
  reviewCount: number;
  tags: string[];
  /** Label of the add-to-cart button. */
  ctaLabel: string;
}

export interface ProductImage {
  /** Path of the image, e.g. "/fe-assets/placeholder-1x1.svg". */
  src: string;
  /** Text alternative for the image. */
  alt: string;
}

function formatPrice(value: number, currency: string): string {
  try {
    return new Intl.NumberFormat(undefined, {
      style: "currency",
      currency,
    }).format(value);
  } catch {
    return `${currency} ${value}`;
  }
}

function formatRating(value: number): string {
  return Number.isFinite(value) ? value.toFixed(1) : "0.0";
}

function formatReviewCount(value: number): string {
  try {
    return new Intl.NumberFormat(undefined, {
      maximumFractionDigits: 0,
    }).format(value);
  } catch {
    return String(value);
  }
}

function StarIcon({ className }: { className?: string }) {
  return (
    <svg
      viewBox="0 0 20 20"
      fill="currentColor"
      aria-hidden="true"
      focusable="false"
      className={className ?? "h-4 w-4"}
    >
      <path d="M10 1.6l2.47 5.01 5.53.8-4 3.9.94 5.5L10 14.2l-4.94 2.6.94-5.5-4-3.9 5.53-.8L10 1.6z" />
    </svg>
  );
}

function StarRating({ rating }: { rating: number }) {
  const safeRating = Number.isFinite(rating)
    ? Math.min(5, Math.max(0, rating))
    : 0;
  const overlayStyle: CSSProperties = {
    width: `${(safeRating / 5) * 100}%`,
  };

  return (
    <div
      className="flex min-w-0 flex-wrap items-center gap-x-2 gap-y-1"
      role="img"
      aria-label={`Rated ${formatRating(rating)} out of 5 stars`}
    >
      <div className="relative inline-flex shrink-0" aria-hidden="true">
        <div className="flex items-center gap-0.5 text-stone-300">
          {[0, 1, 2, 3, 4].map((index) => (
            <StarIcon key={index} className="h-4 w-4 shrink-0" />
          ))}
        </div>
        <div
          className="absolute inset-y-0 left-0 overflow-hidden text-amber-500"
          style={overlayStyle}
        >
          <div className="flex items-center gap-0.5">
            {[0, 1, 2, 3, 4].map((index) => (
              <StarIcon key={index} className="h-4 w-4 shrink-0" />
            ))}
          </div>
        </div>
      </div>
      <span className="min-w-0 break-words text-sm font-semibold tabular-nums text-stone-800">
        {formatRating(rating)}
      </span>
    </div>
  );
}

function ProductCard({
  product,
  currency,
}: {
  product: Product;
  currency: string;
}) {
  const originalPrice = product.originalPrice;
  const onSale = originalPrice !== undefined;
  const ctaLabel = product.ctaLabel.trim() || "Add to cart";

  return (
    <article className="flex h-full min-w-0 flex-col overflow-hidden rounded-2xl border border-stone-200 bg-white shadow-sm transition-shadow hover:shadow-md">
      <div className="relative aspect-square w-full overflow-hidden bg-stone-100">
        <img
          src={product.image.src}
          alt={product.image.alt}
          className="absolute inset-0 h-full w-full object-contain p-3 sm:p-4"
        />
      </div>

      <div className="flex flex-1 flex-col gap-3 p-4 sm:p-5">
        <div className="min-w-0 space-y-2">
          <h2 className="break-words text-base font-semibold leading-snug text-stone-900 [overflow-wrap:anywhere] sm:text-lg">
            {product.name}
          </h2>

          <div className="flex flex-wrap items-baseline gap-x-2 gap-y-1">
            <p className="min-w-0 break-words text-lg font-semibold tabular-nums text-stone-900 [overflow-wrap:anywhere] sm:text-xl">
              {formatPrice(product.price, currency)}
            </p>
            {onSale && originalPrice !== undefined && (
              <>
                <p className="min-w-0 break-words text-sm font-medium tabular-nums text-stone-500 line-through [overflow-wrap:anywhere]">
                  {formatPrice(originalPrice, currency)}
                </p>
                <span className="rounded-full bg-amber-100 px-2 py-0.5 text-[11px] font-semibold uppercase tracking-wide text-amber-900">
                  Sale
                </span>
              </>
            )}
          </div>
        </div>

        <div className="flex min-w-0 flex-wrap items-center gap-x-2 gap-y-1">
          <StarRating rating={product.rating} />
          <span className="min-w-0 break-words text-sm tabular-nums text-stone-600 [overflow-wrap:anywhere]">
            {formatReviewCount(product.reviewCount)}{" "}
            {product.reviewCount === 1 ? "review" : "reviews"}
          </span>
        </div>

        {product.tags.length > 0 && (
          <ul className="flex min-w-0 flex-wrap gap-1.5" aria-label="Product tags">
            {product.tags.map((tag, index) => (
              <li
                key={`${tag}-${index}`}
                className="max-w-full min-w-0 break-words rounded-full bg-stone-100 px-2.5 py-1 text-xs font-medium text-stone-700 [overflow-wrap:anywhere]"
              >
                {tag}
              </li>
            ))}
          </ul>
        )}

        <div className="mt-auto pt-2">
          <button
            type="button"
            aria-label={`${ctaLabel}: ${product.name}`}
            className="block w-full min-w-0 rounded-xl bg-stone-900 px-4 py-3 text-center text-sm font-semibold text-white transition-colors hover:bg-stone-800 focus:outline-none focus-visible:ring-2 focus-visible:ring-amber-500 focus-visible:ring-offset-2"
          >
            <span className="block break-words [overflow-wrap:anywhere]">
              {ctaLabel}
            </span>
          </button>
        </div>
      </div>
    </article>
  );
}

export default function ProductListing({
  heading,
  resultsSummary,
  currency,
  products,
}: Props) {
  return (
    <React.Fragment>
      <section
        aria-labelledby="product-listing-heading"
        className="w-full bg-stone-50 py-10 sm:py-12 lg:py-16"
      >
        <div className="mx-auto w-full max-w-7xl px-4 sm:px-6 lg:px-8">
          <header className="flex min-w-0 flex-col gap-2 border-b border-stone-200 pb-6">
            <h1
              id="product-listing-heading"
              className="min-w-0 break-words text-3xl font-bold tracking-tight text-stone-900 [overflow-wrap:anywhere] sm:text-4xl lg:text-5xl"
            >
              {heading}
            </h1>
            {resultsSummary && (
              <p className="min-w-0 break-words text-sm text-stone-600 [overflow-wrap:anywhere] sm:text-base">
                {resultsSummary}
              </p>
            )}
          </header>

          {products.length > 0 ? (
            <div className="mt-8 grid grid-cols-1 gap-5 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4">
              {products.map((product) => (
                <ProductCard
                  key={product.id}
                  product={product}
                  currency={currency}
                />
              ))}
            </div>
          ) : (
            <div className="mt-8 rounded-2xl border border-dashed border-stone-300 bg-white px-6 py-16 text-center">
              <p className="text-base font-medium text-stone-800">
                No products to show
              </p>
              <p className="mt-1 text-sm text-stone-600">
                Try another category or check back soon.
              </p>
            </div>
          )}
        </div>
      </section>
    </React.Fragment>
  );
}
