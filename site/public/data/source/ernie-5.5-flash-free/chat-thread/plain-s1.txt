import type { FC } from 'react';

export interface Props {
  contact: Contact;
  messages: Message[];
  composer: Composer;
}

export interface Contact {
  name: string;
  /** e.g. "Active now" */
  status: string;
}

/** "me" is the signed-in support agent, "them" the contact. */
export type Sender = 'me' | 'them';

export interface Message {
  id: string;
  from: Sender;
  author: string;
  text: string;
  /** Already formatted, e.g. "9:41 AM". */
  time: string;
}

export interface Composer {
  placeholder: string;
  sendLabel: string;
}

function getInitials(name: string): string {
  const words = name.trim().split(/\s+/).filter(Boolean);

  if (words.length === 0) {
    return '?';
  }

  const first = words[0] ?? '';
  const last = words[words.length - 1] ?? '';

  if (words.length === 1) {
    return first.slice(0, 2).toUpperCase();
  }

  return `${first.charAt(0)}${last.charAt(0)}`.toUpperCase();
}

const ChatThread: FC<Props> = ({ contact, messages, composer }) => {
  const contactInitials = getInitials(contact.name);

  return (
    <section
      aria-label="Chat thread"
      className="mx-auto flex h-[calc(100dvh_-_2rem)] max-h-[680px] w-full max-w-3xl flex-col overflow-hidden rounded-3xl border border-slate-200/80 bg-white shadow-2xl shadow-slate-900/10 ring-1 ring-white/70"
    >
      <header className="flex items-center gap-3 border-b border-slate-200/80 bg-white/95 px-4 py-4 backdrop-blur-sm sm:px-6">
        <div className="relative shrink-0">
          <div className="flex h-11 w-11 items-center justify-center rounded-full bg-linear-to-br from-teal-500 to-cyan-600 text-sm font-semibold tracking-wide text-white shadow-sm shadow-teal-900/20">
            {contactInitials}
          </div>
          <span
            className="absolute -bottom-0.5 -right-0.5 h-3.5 w-3.5 rounded-full border-2 border-white bg-emerald-500"
            aria-hidden="true"
          />
        </div>

        <div className="min-w-0">
          <h1 className="truncate text-base font-semibold text-slate-900">
            {contact.name}
          </h1>
          <p className="mt-0.5 flex items-center gap-1.5 text-xs font-medium text-slate-500">
            <span
              className="h-1.5 w-1.5 rounded-full bg-emerald-500"
              aria-hidden="true"
            />
            {contact.status}
          </p>
        </div>
      </header>

      <div
        className="min-h-0 flex-1 overflow-y-auto bg-slate-50 px-4 py-5 sm:px-6"
        role="log"
        aria-label="Messages"
      >
        <div className="mx-auto flex max-w-2xl flex-col gap-4">
          {messages.length === 0 ? (
            <p className="py-10 text-center text-sm text-slate-400">
              No messages yet.
            </p>
          ) : (
            messages.map((message) => {
              const mine = message.from === 'me';
              const messageInitials = getInitials(
                mine ? message.author : contact.name,
              );

              return (
                <div
                  key={message.id}
                  className={`flex items-end gap-2.5 ${
                    mine ? 'justify-end' : 'justify-start'
                  }`}
                >
                  {!mine && (
                    <div
                      className="flex h-7 w-7 shrink-0 items-center justify-center rounded-full bg-linear-to-br from-teal-500 to-cyan-600 text-[10px] font-bold text-white shadow-sm"
                      aria-hidden="true"
                    >
                      {messageInitials}
                    </div>
                  )}

                  <div
                    className={`flex max-w-[82%] flex-col sm:max-w-[72%] ${
                      mine ? 'items-end text-right' : 'items-start'
                    }`}
                  >
                    <div
                      className={`flex items-baseline gap-2 ${
                        mine ? 'justify-end' : 'justify-start'
                      }`}
                    >
                      <span className="text-xs font-semibold text-slate-600">
                        {message.author}
                      </span>
                      <span className="text-[11px] font-medium tabular-nums text-slate-400">
                        {message.time}
                      </span>
                    </div>

                    <div
                      className={`mt-1.5 rounded-2xl px-3.5 py-2.5 text-sm leading-6 shadow-sm ${
                        mine
                          ? 'rounded-br-md bg-teal-600 text-white shadow-teal-900/10'
                          : 'rounded-bl-md border border-slate-200 bg-white text-slate-700 shadow-slate-900/5'
                      }`}
                    >
                      <p className="whitespace-pre-wrap break-words">
                        {message.text}
                      </p>
                    </div>
                  </div>

                  {mine && (
                    <div
                      className="flex h-7 w-7 shrink-0 items-center justify-center rounded-full bg-slate-800 text-[10px] font-bold text-white shadow-sm"
                      aria-hidden="true"
                    >
                      {messageInitials}
                    </div>
                  )}
                </div>
              );
            })
          )}
        </div>
      </div>

      <footer className="border-t border-slate-200/80 bg-white/95 px-4 py-4 backdrop-blur-sm sm:px-6">
        <div className="mx-auto flex max-w-2xl items-end gap-2.5">
          <div className="flex min-h-[2.75rem] flex-1 items-center rounded-2xl border border-slate-200 bg-slate-50 px-3.5 py-2.5 focus-within:border-teal-500 focus-within:bg-white focus-within:ring-4 focus-within:ring-teal-500/10">
            <input
              type="text"
              readOnly
              placeholder={composer.placeholder}
              aria-label={composer.placeholder}
              className="w-full bg-transparent text-sm text-slate-800 placeholder:text-slate-400 focus:outline-none"
            />
          </div>

          <button
            type="button"
            className="inline-flex h-11 items-center justify-center rounded-xl bg-teal-600 px-4 text-sm font-semibold text-white shadow-sm shadow-teal-900/15 transition hover:bg-teal-700 focus:outline-none focus-visible:ring-4 focus-visible:ring-teal-500/30"
          >
            {composer.sendLabel}
          </button>
        </div>
      </footer>
    </section>
  );
};

export default ChatThread;
