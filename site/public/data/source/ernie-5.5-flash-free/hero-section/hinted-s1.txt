export interface Props {
  /** The hero section at the top of a product landing page. */
  headline: string;
  subheadline: string;
  primaryCtaLabel: string;
  secondaryCtaLabel: string;
  benefits: string[];
  /** Number of customers, e.g. 12400. */
  customerCount: number;
  /** Names of customer companies. */
  customers: string[];
}

/** Formats a number with thousands separators, tolerating any size or edge case. */
function formatNumber(value: number): string {
  if (!Number.isFinite(value)) return String(value);
  const sign = value < 0 ? "-" : "";
  const digits = Math.abs(Math.trunc(value)).toString();
  return sign + digits.replace(/\B(?=(\d{3})+(?!\d))/g, ",");
}

/** Small inline check icon used to decorate each benefit. */
function CheckIcon() {
  return (
    <svg
      className="h-5 w-5 shrink-0 text-indigo-600"
      viewBox="0 0 20 20"
      fill="currentColor"
      aria-hidden="true"
    >
      <path
        fillRule="evenodd"
        d="M16.704 4.153a.75.75 0 0 1 .143 1.052l-8 10.5a.75.75 0 0 1-1.127.075l-4.5-4.5a.75.75 0 0 1 1.06-1.06l3.894 3.893 7.48-9.817a.75.75 0 0 1 1.05-.143Z"
        clipRule="evenodd"
      />
    </svg>
  );
}

export default function Hero({
  headline,
  subheadline,
  primaryCtaLabel,
  secondaryCtaLabel,
  benefits,
  customerCount,
  customers,
}: Props) {
  const showBenefits = benefits.length > 0;
  const showCustomers = customers.length > 0;

  return (
    <section className="w-full bg-gradient-to-b from-slate-50 to-white px-5 py-16 sm:px-8 sm:py-20 lg:py-28">
      <div className="mx-auto flex max-w-4xl flex-col items-center text-center">
        <h1 className="max-w-3xl text-balance text-4xl font-bold tracking-tight text-slate-900 break-words sm:text-5xl lg:text-6xl">
          {headline}
        </h1>

        <p className="mt-6 max-w-2xl text-pretty text-lg leading-relaxed text-slate-600 break-words sm:text-xl">
          {subheadline}
        </p>

        <div className="mt-8 flex w-full flex-col items-stretch gap-3 sm:w-auto sm:flex-row sm:items-center">
          <button
            type="button"
            className="inline-flex items-center justify-center rounded-lg bg-indigo-600 px-7 py-3.5 text-base font-semibold text-white shadow-lg shadow-indigo-600/30 transition-colors hover:bg-indigo-700 focus:outline-none focus-visible:ring-2 focus-visible:ring-indigo-500 focus-visible:ring-offset-2"
          >
            {primaryCtaLabel}
          </button>
          <button
            type="button"
            className="inline-flex items-center justify-center rounded-lg border border-slate-300 bg-white px-7 py-3.5 text-base font-semibold text-slate-700 transition-colors hover:bg-slate-50 focus:outline-none focus-visible:ring-2 focus-visible:ring-slate-400 focus-visible:ring-offset-2"
          >
            {secondaryCtaLabel}
          </button>
        </div>

        {showBenefits && (
          <ul className="mt-10 flex flex-wrap items-center justify-center gap-x-6 gap-y-3 text-sm font-medium text-slate-600">
            {benefits.map((benefit, index) => (
              <li
                key={index}
                className="flex min-w-0 items-center gap-2 break-words"
              >
                <CheckIcon />
                <span className="min-w-0 break-words">{benefit}</span>
              </li>
            ))}
          </ul>
        )}

        <div className="mt-16 w-full">
          <p className="text-sm font-medium text-slate-500 break-words">
            Trusted by {formatNumber(customerCount)}{" "}
            {customerCount === 1 ? "support team" : "support teams"} around the
            world
          </p>

          {showCustomers && (
            <div className="mt-6 flex flex-wrap items-center justify-center gap-x-10 gap-y-6">
              {customers.map((customer, index) => (
                <span
                  key={index}
                  className="max-w-full text-xl font-semibold tracking-tight text-slate-400 break-words sm:text-2xl"
                >
                  {customer}
                </span>
              ))}
            </div>
          )}
        </div>
      </div>
    </section>
  );
}
