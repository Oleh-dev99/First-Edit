function wordmarkClass(index: number): string {
  const styles = [
    "font-serif text-xl font-semibold tracking-[-0.03em] sm:text-2xl",
    "text-sm font-semibold uppercase tracking-[0.22em] sm:text-base",
    "font-mono text-lg font-medium tracking-[0.04em] sm:text-xl",
    "font-serif text-xl font-medium italic sm:text-2xl",
    "text-base font-black uppercase tracking-[0.12em] sm:text-lg",
    "font-serif text-xl font-bold tracking-[0.02em] sm:text-2xl",
  ] as const;

  return styles[index % styles.length] ?? "text-lg font-semibold";
}
