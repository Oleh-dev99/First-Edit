import { useId } from 'react';

export type MetricFormat = 'currency' | 'percent' | 'count';

export interface Metric {
  id: string;
  label: string;
  /** Raw value. For "percent", a fraction: 0.034 means 3.4%. */
  value: number;
  format: MetricFormat;
  /** Change against the previous period, as a fraction: -0.052 means -5.2%. */
  change: number;
  caption: string;
}

/** The key metric tiles at the top of a store analytics dashboard. */
export interface Props {
  title: string;
  /** The period the metrics cover, e.g. "Last 30 days". */
  periodLabel: string;
  /** ISO 4217 code used for currency values, e.g. "USD". */
  currency: string;
  metrics: Metric[];
}

type Direction = 'up' | 'down' | 'flat';

const LOCALE = 'en-US';

const BADGE_STYLES: Record<Direction, string> = {
  up: 'bg-emerald-50 text-emerald-700 ring-1 ring-inset ring-emerald-600/20',
  down: 'bg-rose-50 text-rose-700 ring-1 ring-inset ring-rose-600/20',
  flat: 'bg-slate-100 text-slate-600 ring-1 ring-inset ring-slate-500/20',
};

const ACCENT_STYLES: Record<Direction, string> = {
  up: 'bg-emerald-400/80',
  down: 'bg-rose-400/80',
  flat: 'bg-slate-300',
};

const DIRECTION_WORD: Record<Direction, string> = {
  up: 'Increased',
  down: 'Decreased',
  flat: 'No change',
};

function safeFormat(value: number, options: Intl.NumberFormatOptions): string {
  if (!Number.isFinite(value)) return '—';
  try {
    return new Intl.NumberFormat(LOCALE, options).format(value);
  } catch {
    return String(value);
  }
}

function formatCurrency(value: number, currency: string): string {
  const digits = Math.abs(value) >= 1000 ? 0 : 2;
  try {
    return new Intl.NumberFormat(LOCALE, {
      style: 'currency',
      currency,
      currencyDisplay: 'narrowSymbol',
      minimumFractionDigits: digits,
      maximumFractionDigits: digits,
    }).format(value);
  } catch {
    return `${currency} ${safeFormat(value, {
      minimumFractionDigits: digits,
      maximumFractionDigits: digits,
    })}`;
  }
}

function formatValue(metric: Metric, currency: string): string {
  switch (metric.format) {
    case 'currency':
      return formatCurrency(metric.value, currency);
    case 'percent':
      return `${safeFormat(metric.value * 100, {
        minimumFractionDigits: 0,
        maximumFractionDigits: 1,
      })}%`;
    case 'count':
    default:
      return safeFormat(metric.value, { maximumFractionDigits: 0 });
  }
}

function roundToTenth(value: number): number {
  return Math.round(value * 10) / 10;
}

function directionOf(change: number): Direction {
  const pct = roundToTenth(change * 100);
  if (pct > 0) return 'up';
  if (pct < 0) return 'down';
  return 'flat';
}

function formatChange(change: number): string {
  const pct = roundToTenth(change * 100);
  if (pct === 0) return '0%';
  const body = safeFormat(Math.abs(pct), {
    minimumFractionDigits: 1,
    maximumFractionDigits: 1,
  });
  return `${pct > 0 ? '+' : '−'}${body}%`;
}

function TrendIcon({ direction }: { direction: Direction }) {
  if (direction === 'flat') {
    return (
      <svg
        aria-hidden="true"
        viewBox="0 0 12 12"
        fill="none"
        stroke="currentColor"
        strokeWidth={1.75}
        strokeLinecap="round"
        className="h-3 w-3"
      >
        <path d="M2.5 6h7" />
      </svg>
    );
  }

  const path =
    direction === 'up'
      ? 'M6 10V2.6M6 2.6 2.9 5.7M6 2.6 9.1 5.7'
      : 'M6 2v7.4M6 9.4 2.9 6.3M6 9.4 9.1 6.3';

  return (
    <svg
      aria-hidden="true"
      viewBox="0 0 12 12"
      fill="none"
      stroke="currentColor"
      strokeWidth={1.75}
      strokeLinecap="round"
      strokeLinejoin="round"
      className="h-3 w-3"
    >
      <path d={path} />
    </svg>
  );
}

function CalendarIcon() {
  return (
    <svg
      aria-hidden="true"
      viewBox="0 0 16 16"
      fill="none"
      stroke="currentColor"
      strokeWidth={1.4}
      strokeLinecap="round"
      strokeLinejoin="round"
      className="h-3.5 w-3.5"
    >
      <rect x="2.25" y="3.75" width="11.5" height="10" rx="2" />
      <path d="M2.25 6.75h11.5M5.5 2.25v2.5M10.5 2.25v2.5" />
    </svg>
  );
}

function MetricTile({ metric, currency }: { metric: Metric; currency: string }) {
  const direction = directionOf(metric.change);

  return (
    <article className="relative flex h-full flex-col overflow-hidden rounded-2xl border border-slate-200 bg-white p-5 shadow-sm transition-colors hover:border-slate-300">
      <span
        aria-hidden="true"
        className={`absolute inset-x-0 top-0 h-[3px] ${ACCENT_STYLES[direction]}`}
      />

      <h3 className="text-[0.7rem] font-semibold uppercase tracking-[0.12em] text-slate-500">
        {metric.label}
      </h3>

      <p className="mt-2 text-3xl font-semibold tracking-tight text-slate-900 tabular-nums">
        {formatValue(metric, currency)}
      </p>

      <div className="mt-3 flex flex-wrap items-center gap-x-2 gap-y-1">
        <span
          className={`inline-flex items-center gap-1 rounded-full px-2 py-0.5 text-xs font-semibold tabular-nums ${BADGE_STYLES[direction]}`}
        >
          <TrendIcon direction={direction} />
          <span className="sr-only">{DIRECTION_WORD[direction]} </span>
          {formatChange(metric.change)}
        </span>
        <span className="text-xs text-slate-400">vs previous period</span>
      </div>

      <p className="mt-auto pt-4 text-sm leading-relaxed text-slate-500">
        {metric.caption}
      </p>
    </article>
  );
}

export default function MetricTiles({
  title,
  periodLabel,
  currency,
  metrics,
}: Props) {
  const headingId = useId();

  return (
    <section
      aria-labelledby={headingId}
      className="w-full border-b border-slate-200 bg-slate-50 px-4 py-8 sm:px-6 lg:px-8 lg:py-10"
    >
      <div className="mx-auto w-full max-w-7xl">
        <header className="flex flex-col gap-4 sm:flex-row sm:items-end sm:justify-between">
          <div>
            <h2
              id={headingId}
              className="text-xl font-semibold tracking-tight text-slate-900 sm:text-2xl"
            >
              {title}
            </h2>
            <p className="mt-1 text-sm text-slate-500">
              Change is measured against the previous period.
            </p>
          </div>

          <div className="inline-flex w-fit items-center gap-2 rounded-full border border-slate-200 bg-white px-3.5 py-1.5 text-xs font-medium text-slate-600 shadow-sm">
            <span className="text-slate-400">
              <CalendarIcon />
            </span>
            <span>{periodLabel}</span>
            <span aria-hidden="true" className="text-slate-300">
              /
            </span>
            <span className="font-semibold uppercase tracking-wide text-slate-700">
              {currency}
            </span>
          </div>
        </header>

        {metrics.length > 0 ? (
          <div className="mt-6 grid grid-cols-1 gap-4 sm:grid-cols-2 lg:gap-5 xl:grid-cols-4">
            {metrics.map((metric, index) => (
              <MetricTile
                key={metric.id || `${metric.label}-${index}`}
                metric={metric}
                currency={currency}
              />
            ))}
          </div>
        ) : (
          <p className="mt-6 rounded-2xl border border-dashed border-slate-300 bg-white/60 px-5 py-10 text-center text-sm text-slate-500">
            No metrics are available for {periodLabel}.
          </p>
        )}
      </div>
    </section>
  );
}
