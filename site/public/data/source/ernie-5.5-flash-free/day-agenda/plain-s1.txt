export interface Props {
  dateLabel: string;
  events: AgendaEvent[];
  addEventLabel: string;
}

export type TagColor = 'blue' | 'green' | 'amber' | 'red' | 'purple' | 'gray';

export interface EventTag {
  label: string;
  color: TagColor;
}

export interface AgendaEvent {
  id: string;
  start: string;
  end: string;
  title: string;
  location: string;
  attendees: string[];
  tag: EventTag;
}

const TAG_STYLES: Record<TagColor, { badge: string; dot: string }> = {
  blue: {
    badge: "border-blue-200 bg-blue-50 text-blue-700",
    dot: "bg-blue-500",
  },
  green: {
    badge: "border-emerald-200 bg-emerald-50 text-emerald-700",
    dot: "bg-emerald-500",
  },
  amber: {
    badge: "border-amber-200 bg-amber-50 text-amber-800",
    dot: "bg-amber-500",
  },
  red: {
    badge: "border-red-200 bg-red-50 text-red-700",
    dot: "bg-red-500",
  },
  purple: {
    badge: "border-purple-200 bg-purple-50 text-purple-700",
    dot: "bg-purple-500",
  },
  gray: {
    badge: "border-slate-200 bg-slate-50 text-slate-700",
    dot: "bg-slate-400",
  },
};

export default function DayAgenda({ dateLabel, events, addEventLabel }: Props) {
  return (
    <section
      aria-label={dateLabel}
      className="w-full max-w-2xl overflow-hidden rounded-3xl border border-slate-200 bg-white shadow-sm"
    >
      <header className="flex flex-col gap-4 border-b border-slate-100 px-5 py-5 sm:flex-row sm:items-center sm:justify-between sm:px-6">
        <div className="min-w-0">
          <h1 className="text-2xl font-semibold tracking-tight text-slate-900">{dateLabel}</h1>
          <p className="mt-1 text-sm text-slate-500">Your day at a glance</p>
        </div>

        <button
          type="button"
          className="inline-flex items-center justify-center gap-2 self-start rounded-full bg-slate-900 px-4 py-2.5 text-sm font-semibold text-white shadow-sm outline-none transition hover:bg-slate-800 focus-visible:ring-2 focus-visible:ring-slate-900 focus-visible:ring-offset-2 sm:self-auto"
        >
          <span
            aria-hidden="true"
            className="grid h-5 w-5 place-items-center rounded-full bg-white/10 text-base leading-none"
          >
            +
          </span>
          {addEventLabel}
        </button>
      </header>

      <div className="px-5 py-5 sm:px-6 sm:py-6">
        {events.length === 0 ? (
          <div className="rounded-2xl border border-dashed border-slate-300 bg-slate-50/60 px-5 py-10 text-center">
            <p className="text-sm font-medium text-slate-700">No events for this day</p>
            <p className="mt-1 text-sm text-slate-500">Your calendar is wide open.</p>
          </div>
        ) : (
          <ol className="space-y-3">
            {events.map((event) => {
              const tagStyle = TAG_STYLES[event.tag.color];

              return (
                <li
                  key={event.id}
                  className="rounded-2xl border border-slate-200 bg-white p-4 transition hover:border-slate-300 hover:shadow-sm sm:p-5"
                >
                  <article className="flex flex-col gap-4 sm:flex-row sm:gap-5">
                    <div className="flex shrink-0 items-center gap-2 text-sm font-medium text-slate-900 sm:w-24 sm:flex-col sm:items-start sm:gap-0.5 sm:border-r sm:border-slate-100 sm:pr-4">
                      <span>{event.start}</span>
                      <span aria-hidden="true" className="text-slate-300 sm:hidden">
                        –
                      </span>
                      <span className="text-slate-500">{event.end}</span>
                    </div>

                    <div className="min-w-0 flex-1">
                      <div className="flex flex-wrap items-start justify-between gap-x-3 gap-y-2">
                        <h2 className="min-w-0 text-base font-semibold leading-6 text-slate-900 sm:text-lg">
                          {event.title}
                        </h2>

                        <span
                          className={`inline-flex shrink-0 items-center gap-1.5 rounded-full border px-2.5 py-1 text-xs font-medium ${tagStyle.badge}`}
                        >
                          <span className={`h-1.5 w-1.5 rounded-full ${tagStyle.dot}`} aria-hidden="true" />
                          {event.tag.label}
                        </span>
                      </div>

                      <div className="mt-2 space-y-1 text-sm text-slate-600">
                        {event.location && (
                          <p className="flex flex-wrap items-baseline gap-x-2 gap-y-0.5">
                            <span className="font-medium text-slate-500">Location</span>
                            <span aria-hidden="true" className="text-slate-300">
                              ·
                            </span>
                            <span className="break-words">{event.location}</span>
                          </p>
                        )}

                        {event.attendees.length > 0 && (
                          <p className="flex flex-wrap items-baseline gap-x-2 gap-y-0.5">
                            <span className="font-medium text-slate-500">With</span>
                            <span aria-hidden="true" className="text-slate-300">
                              ·
                            </span>
                            <span className="break-words">{event.attendees.join(", ")}</span>
                          </p>
                        )}
                      </div>
                    </div>
                  </article>
                </li>
              );
            })}
          </ol>
        )}
      </div>
    </section>
  );
}
