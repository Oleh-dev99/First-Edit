import * as React from "react";

export interface Props {
  columns: Column[];
}

export interface Column {
  id: string;
  title: string;
  cards: Card[];
}

export interface Card {
  id: string;
  title: string;
  tags: string[];
  /** The assignee's name. */
  assignee: string;
  /** Already formatted, e.g. "Nov 18". */
  dueDate: string;
  commentCount: number;
}

const AVATAR_STYLES = [
  "bg-rose-100 text-rose-700 ring-rose-200",
  "bg-amber-100 text-amber-700 ring-amber-200",
  "bg-emerald-100 text-emerald-700 ring-emerald-200",
  "bg-sky-100 text-sky-700 ring-sky-200",
  "bg-violet-100 text-violet-700 ring-violet-200",
  "bg-teal-100 text-teal-700 ring-teal-200",
];

function getInitials(name: string): string {
  const parts = name.trim().split(/\s+/).filter(Boolean);

  if (parts.length === 0) {
    return "?";
  }

  const first = parts[0] ?? "";
  const last = parts.length > 1 ? parts[parts.length - 1] ?? "" : "";
  const firstInitial = Array.from(first)[0] ?? "";
  const lastInitial = Array.from(last)[0] ?? "";

  return `${firstInitial}${lastInitial}`.toUpperCase() || "?";
}

function getAvatarStyle(name: string): string {
  let hash = 0;

  for (let index = 0; index < name.length; index += 1) {
    hash = (hash * 31 + name.charCodeAt(index)) | 0;
  }

  const styleIndex = Math.abs(hash) % AVATAR_STYLES.length;

  return (
    AVATAR_STYLES[styleIndex] ??
    "bg-slate-100 text-slate-700 ring-slate-200"
  );
}

function CalendarIcon() {
  return (
    <svg
      viewBox="0 0 24 24"
      fill="none"
      stroke="currentColor"
      strokeWidth="2"
      strokeLinecap="round"
      strokeLinejoin="round"
      aria-hidden="true"
      focusable="false"
      className="h-3.5 w-3.5 shrink-0 text-slate-400"
    >
      <rect x="3" y="4" width="18" height="18" rx="2" />
      <path d="M16 2v4M8 2v4M3 10h18" />
    </svg>
  );
}

function CommentIcon() {
  return (
    <svg
      viewBox="0 0 24 24"
      fill="none"
      stroke="currentColor"
      strokeWidth="2"
      strokeLinecap="round"
      strokeLinejoin="round"
      aria-hidden="true"
      focusable="false"
      className="h-3.5 w-3.5 shrink-0 text-slate-400"
    >
      <path d="M21 15a4 4 0 0 1-4 4H8l-5 3V7a4 4 0 0 1 4-4h10a4 4 0 0 1 4 4z" />
    </svg>
  );
}

export default function ProjectBoard({ columns }: Props) {
  const headingId = React.useId();
  const totalCards = columns.reduce(
    (total, column) => total + column.cards.length,
    0,
  );

  return (
    <section
      aria-labelledby={headingId}
      className="w-full border-t border-slate-200/70 bg-slate-50 px-3 py-6 sm:px-6 sm:py-8 lg:px-8"
    >
      <div className="mx-auto w-full max-w-[100rem]">
        <div className="mb-4 min-w-0 sm:mb-5">
          <h2
            id={headingId}
            className="break-words text-lg font-semibold tracking-tight text-slate-900 sm:text-xl"
          >
            Task board
          </h2>
          <p className="mt-1 break-words text-sm text-slate-500">
            {totalCards} {totalCards === 1 ? "task" : "tasks"} across{" "}
            {columns.length} {columns.length === 1 ? "column" : "columns"}
          </p>
        </div>

        {columns.length === 0 ? (
          <div className="rounded-2xl border border-dashed border-slate-300 bg-white/70 px-4 py-10 text-center text-sm text-slate-500">
            No columns to show.
          </div>
        ) : (
          <ol
            className="m-0 grid min-w-0 list-none gap-3 p-0 sm:gap-4"
            style={{
              gridTemplateColumns:
                "repeat(auto-fit, minmax(min(100%, 17rem), 1fr))",
            }}
          >
            {columns.map((column, columnIndex) => {
              const cardCount = column.cards.length;

              return (
                <li key={`${column.id}-${columnIndex}`} className="flex min-w-0">
                  <section
                    aria-label={column.title}
                    className="flex h-full w-full min-w-0 flex-col rounded-2xl border border-slate-200/80 bg-slate-100/70 p-3 shadow-sm"
                  >
                    <div className="mb-3 flex flex-wrap items-center justify-between gap-2 px-0.5">
                      <h3 className="min-w-0 break-words text-sm font-semibold tracking-tight text-slate-900">
                        {column.title}
                      </h3>
                      <span className="inline-flex h-6 min-w-6 shrink-0 items-center justify-center rounded-full border border-slate-200 bg-white px-2 text-xs font-semibold text-slate-500 shadow-sm">
                        <span aria-hidden="true">{cardCount}</span>
                        <span className="sr-only">
                          {cardCount} {cardCount === 1 ? "card" : "cards"}
                        </span>
                      </span>
                    </div>

                    {cardCount === 0 ? (
                      <div className="rounded-xl border border-dashed border-slate-300/80 bg-white/50 px-3 py-6 text-center text-xs font-medium text-slate-400">
                        No tasks yet
                      </div>
                    ) : (
                      <ul className="m-0 list-none space-y-2.5 p-0">
                        {column.cards.map((card, cardIndex) => {
                          const assigneeName = card.assignee.trim();
                          const displayName = assigneeName || "Unassigned";
                          const avatarStyle = getAvatarStyle(displayName);
                          const initials = getInitials(displayName);
                          const tags = card.tags.filter(
                            (tag) => tag.trim().length > 0,
                          );

                          return (
                            <li key={`${card.id}-${cardIndex}`} className="min-w-0">
                              <article className="flex min-w-0 flex-col gap-3 rounded-xl border border-slate-200/90 bg-white p-3.5 shadow-sm sm:p-4">
                                <h4 className="break-words text-sm font-semibold leading-snug text-slate-900">
                                  {card.title}
                                </h4>

                                {tags.length > 0 && (
                                  <div className="flex max-w-full flex-wrap gap-1.5">
                                    {tags.map((tag, tagIndex) => (
                                      <span
                                        key={`${tag}-${tagIndex}`}
                                        className="inline-block max-w-full break-words rounded-full bg-slate-100 px-2 py-0.5 text-[11px] font-medium leading-4 text-slate-600 ring-1 ring-inset ring-slate-200"
                                      >
                                        {tag}
                                      </span>
                                    ))}
                                  </div>
                                )}

                                <div className="flex min-w-0 items-center gap-2">
                                  <span
                                    aria-hidden="true"
                                    className={`flex h-7 w-7 shrink-0 items-center justify-center rounded-full text-[10px] font-bold uppercase tracking-wide ring-1 ring-inset ${avatarStyle}`}
                                  >
                                    {initials}
                                  </span>
                                  <span className="min-w-0 flex-1 break-words text-xs font-medium leading-5 text-slate-600">
                                    <span className="sr-only">Assigned to </span>
                                    {displayName}
                                  </span>
                                </div>

                                <div className="flex flex-wrap items-center justify-between gap-x-3 gap-y-2 border-t border-slate-100 pt-3 text-xs text-slate-500">
                                  <span className="inline-flex min-w-0 max-w-full items-center gap-1.5">
                                    <CalendarIcon />
                                    <span className="sr-only">Due date: </span>
                                    <span className="min-w-0 break-words">
                                      {card.dueDate || "No due date"}
                                    </span>
                                  </span>

                                  <span className="inline-flex min-w-0 max-w-full items-center gap-1.5">
                                    <CommentIcon />
                                    <span className="min-w-0 break-words tabular-nums">
                                      {card.commentCount}
                                    </span>
                                    <span className="sr-only">
                                      {card.commentCount === 1
                                        ? " comment"
                                        : " comments"}
                                    </span>
                                  </span>
                                </div>
                              </article>
                            </li>
                          );
                        })}
                      </ul>
                    )}
                  </section>
                </li>
              );
            })}
          </ol>
        )}
      </div>
    </section>
  );
}
