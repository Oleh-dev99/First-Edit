/**
 * Inbox message list for Quillpost.
 * Renders the folder header plus one row per message.
 */
export interface InboxMessage {
  id: string;
  sender: string;
  subject: string;
  /** The opening text of the message. */
  preview: string;
  /** Already formatted, e.g. "9:41 AM" or "Yesterday". */
  time: string;
  unread: boolean;
  labels: string[];
  hasAttachment: boolean;
}

/** The message list of a web email client's inbox. */
export interface Props {
  folderTitle: string;
  unreadCount: number;
  messages: InboxMessage[];
}

function initials(name: string): string {
  return name
    .split(/\s+/)
    .filter(Boolean)
    .slice(0, 2)
    .map((part) => part.charAt(0).toUpperCase())
    .join("");
}

function PaperclipIcon() {
  return (
    <svg
      aria-hidden="true"
      viewBox="0 0 24 24"
      fill="none"
      stroke="currentColor"
      strokeWidth="1.8"
      strokeLinecap="round"
      strokeLinejoin="round"
      className="h-3.5 w-3.5"
    >
      <path d="M21.44 11.05l-9.19 9.19a6 6 0 0 1-8.49-8.49l8.49-8.49a4 4 0 1 1 5.66 5.66l-8.49 8.49a2 2 0 0 1-2.83-2.83l8.49-8.49" />
    </svg>
  );
}

function UnreadDot() {
  return (
    <span
      aria-hidden="true"
      className="inline-block h-2 w-2 shrink-0 rounded-full bg-blue-600"
    />
  );
}

export default function InboxMessageList({
  folderTitle,
  unreadCount,
  messages,
}: Props) {
  const hasUnread = unreadCount > 0;

  return (
    <section
      aria-labelledby="folder-title"
      className="w-full min-w-0 bg-white"
    >
      {/* Folder header */}
      <div className="flex flex-wrap items-end justify-between gap-3 border-b border-slate-200 px-5 py-5 sm:px-8">
        <div className="min-w-0">
          <h1
            id="folder-title"
            className="truncate text-2xl font-semibold tracking-tight text-slate-900"
          >
            {folderTitle}
          </h1>
          <p className="mt-1 text-sm text-slate-500">
            {hasUnread ? (
              <>
                <span className="font-semibold text-blue-700">
                  {unreadCount} unread
                </span>
                <span className="text-slate-400"> · </span>
                {messages.length} message{messages.length === 1 ? "" : "s"}
              </>
            ) : (
              <>{messages.length} message{messages.length === 1 ? "" : "s"}</>
            )}
          </p>
        </div>

        {hasUnread && (
          <span className="inline-flex items-center gap-1.5 rounded-full bg-blue-50 px-3 py-1 text-xs font-semibold text-blue-700 ring-1 ring-inset ring-blue-200">
            <span className="h-1.5 w-1.5 rounded-full bg-blue-600" />
            {unreadCount} unread
          </span>
        )}
      </div>

      {/* Messages */}
      {messages.length === 0 ? (
        <p className="px-5 py-16 text-center text-sm text-slate-500 sm:px-8">
          No messages in this folder.
        </p>
      ) : (
        <ul className="divide-y divide-slate-200">
          {messages.map((message) => {
            const rowTint = message.unread
              ? "bg-blue-50/70 hover:bg-blue-50"
              : "bg-white hover:bg-slate-50";

            return (
              <li key={message.id} className="relative">
                <div
                  className={`flex min-w-0 items-start gap-3 px-5 py-4 transition-colors sm:gap-4 sm:px-8 ${rowTint}`}
                >
                  {/* Unread accent bar */}
                  {message.unread && (
                    <span
                      aria-hidden="true"
                      className="absolute inset-y-0 left-0 w-1 bg-blue-600"
                    />
                  )}

                  {/* Avatar */}
                  <span
                    aria-hidden="true"
                    className={`mt-0.5 flex h-10 w-10 shrink-0 items-center justify-center rounded-full text-xs font-semibold ring-1 ring-inset ${
                      message.unread
                        ? "bg-blue-600/10 text-blue-700 ring-blue-600/20"
                        : "bg-slate-100 text-slate-500 ring-slate-200"
                    }`}
                  >
                    {initials(message.sender)}
                  </span>

                  <div className="min-w-0 flex-1">
                    <div className="flex min-w-0 items-start justify-between gap-4">
                      <div className="min-w-0 flex-1">
                        {/* Sender */}
                        <div className="flex min-w-0 items-center gap-2">
                          {message.unread && <UnreadDot />}
                          <p
                            className={`truncate text-sm ${
                              message.unread
                                ? "font-semibold text-slate-900"
                                : "font-medium text-slate-600"
                            }`}
                          >
                            {message.sender}
                          </p>
                          {message.hasAttachment && (
                            <span
                              className="inline-flex shrink-0 items-center gap-1 rounded border border-slate-300 bg-white px-1.5 py-0.5 text-[11px] font-medium text-slate-500"
                              title="Has attachment"
                              aria-label="Has attachment"
                            >
                              <PaperclipIcon />
                              <span className="sr-only">Attachment</span>
                            </span>
                          )}
                        </div>

                        {/* Subject */}
                        <p
                          className={`mt-0.5 truncate ${
                            message.unread
                              ? "text-[15px] font-semibold text-slate-900"
                              : "text-[15px] font-medium text-slate-700"
                          }`}
                        >
                          {message.subject}
                        </p>
                      </div>

                      {/* Time — shown exactly as provided */}
                      <time
                        className={`shrink-0 pt-0.5 text-xs tabular-nums ${
                          message.unread
                            ? "font-semibold text-blue-700"
                            : "text-slate-400"
                        }`}
                      >
                        {message.time}
                      </time>
                    </div>

                    {/* Preview */}
                    <p
                      className={`mt-1 line-clamp-2 text-sm ${
                        message.unread ? "text-slate-600" : "text-slate-400"
                      }`}
                    >
                      {message.preview}
                    </p>

                    {/* Labels */}
                    {message.labels.length > 0 && (
                      <ul className="mt-2.5 flex flex-wrap items-center gap-1.5">
                        {message.labels.map((label) => (
                          <li
                            key={label}
                            className={`rounded-full px-2 py-0.5 text-[11px] font-medium ring-1 ring-inset ${
                              message.unread
                                ? "bg-white text-slate-600 ring-slate-300"
                                : "bg-slate-50 text-slate-500 ring-slate-200"
                            }`}
                          >
                            {label}
                          </li>
                        ))}
                      </ul>
                    )}
                  </div>
                </div>
              </li>
            );
          })}
        </ul>
      )}
    </section>
  );
}
