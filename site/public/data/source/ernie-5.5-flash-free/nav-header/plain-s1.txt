import * as React from "react";

export interface Props {
  /** The top navigation bar of a web app. */
  productName: string;
  links: NavLink[];
  /** Label of the primary call-to-action button. */
  ctaLabel: string;
  /** The signed-in user. */
  user: User;
}

export interface NavLink {
  label: string;
  /** Where the link points, e.g. "/reports". */
  href: string;
  /** True for the link to the current page. */
  active: boolean;
}

export interface User {
  name: string;
  email: string;
}

function getInitials(name: string): string {
  return name
    .trim()
    .split(/\s+/)
    .filter(Boolean)
    .slice(0, 2)
    .map((part) => part.charAt(0).toUpperCase())
    .join("");
}

export default function AppNavigationHeader({
  productName,
  links,
  ctaLabel,
  user,
}: Props): React.ReactElement {
  const initials = getInitials(user.name);
  const brandInitial = productName.trim().charAt(0).toUpperCase() || "S";

  return (
    <header className="w-full border-b border-stone-200/80 bg-[#fbfaf7]">
      <div className="mx-auto flex w-full max-w-[1480px] flex-wrap items-center gap-x-4 gap-y-3 px-4 py-3 sm:px-6 lg:min-h-[72px] lg:flex-nowrap lg:gap-x-6 lg:px-8 lg:py-0">
        <a
          href="/"
          aria-label={`${productName} home`}
          className="group order-1 flex shrink-0 items-center gap-2.5 rounded-xl focus:outline-none focus-visible:ring-2 focus-visible:ring-emerald-700 focus-visible:ring-offset-2 lg:order-none"
        >
          <span className="relative flex h-9 w-9 items-center justify-center rounded-[12px] bg-emerald-950 font-serif text-lg font-semibold text-white shadow-sm ring-1 ring-emerald-900/20 transition-transform group-hover:-translate-y-0.5">
            {brandInitial}
            <span
              aria-hidden="true"
              className="absolute -right-1 -top-1 h-2.5 w-2.5 rounded-full bg-lime-300 ring-2 ring-[#fbfaf7]"
            />
          </span>
          <span className="font-serif text-[1.35rem] font-semibold tracking-[-0.02em] text-stone-950">
            {productName}
          </span>
        </a>

        <nav
          aria-label="Primary"
          className="order-3 -mx-1 w-full overflow-x-auto border-t border-stone-200/80 px-1 pt-3 lg:order-none lg:mx-0 lg:w-auto lg:flex-1 lg:overflow-visible lg:border-t-0 lg:px-0 lg:pt-0"
        >
          <ul className="flex min-w-max items-center gap-1 lg:min-w-0 lg:flex-wrap">
            {links.map((link) => (
              <li key={`${link.href}-${link.label}`}>
                <a
                  href={link.href}
                  aria-current={link.active ? "page" : undefined}
                  className={`inline-flex items-center gap-2 rounded-xl px-3 py-2 text-sm font-medium transition-colors focus:outline-none focus-visible:ring-2 focus-visible:ring-emerald-700 focus-visible:ring-offset-2 ${
                    link.active
                      ? "bg-emerald-50 text-emerald-950 ring-1 ring-inset ring-emerald-200"
                      : "text-stone-600 hover:bg-stone-100 hover:text-stone-950"
                  }`}
                >
                  <span
                    aria-hidden="true"
                    className={`h-1.5 w-1.5 rounded-full ${
                      link.active ? "bg-emerald-600" : "bg-transparent"
                    }`}
                  />
                  <span>{link.label}</span>
                </a>
              </li>
            ))}
          </ul>
        </nav>

        <div className="order-2 ml-auto flex items-center gap-3 lg:order-none">
          <button
            type="button"
            className="inline-flex items-center justify-center gap-2 rounded-xl bg-emerald-950 px-4 py-2.5 text-sm font-semibold text-white shadow-sm ring-1 ring-inset ring-emerald-950/20 transition hover:bg-emerald-900 focus:outline-none focus-visible:ring-2 focus-visible:ring-emerald-700 focus-visible:ring-offset-2"
          >
            <span
              aria-hidden="true"
              className="flex h-5 w-5 items-center justify-center rounded-full bg-white/10 text-base leading-none"
            >
              +
            </span>
            <span>{ctaLabel}</span>
          </button>

          <span
            aria-hidden="true"
            className="hidden h-8 w-px bg-stone-200 sm:block"
          />

          <div className="flex min-w-0 items-center gap-3">
            <span
              aria-hidden="true"
              className="flex h-10 w-10 shrink-0 items-center justify-center rounded-full bg-amber-100 text-sm font-semibold tracking-wide text-amber-950 ring-1 ring-inset ring-amber-900/10"
            >
              {initials || "U"}
            </span>
            <span className="min-w-0">
              <span className="block truncate text-sm font-semibold leading-tight text-stone-900">
                {user.name}
              </span>
              <span
                className="block truncate text-xs leading-tight text-stone-500"
                title={user.email}
              >
                {user.email}
              </span>
            </span>
          </div>
        </div>
      </div>
    </header>
  );
}
